  // Note: We disallow negative values as this is used for timestamp queries
  // where it's possible for a query to return a beginning time greater than the
  // end time. See: https://gpuweb.github.io/gpuweb/#timestamp
  class NonNegativeRollingAverage {
    #total = 0;
    #samples = [];
    #cursor = 0;
    #numSamples;
    constructor(numSamples = 30) {
      this.#numSamples = numSamples;
    }
    addSample(v) {
      if (!Number.isNaN(v) && Number.isFinite(v) && v >= 0) {
        this.#total += v - (this.#samples[this.#cursor] || 0);
        this.#samples[this.#cursor] = v;
        this.#cursor = (this.#cursor + 1) % this.#numSamples;
      }
    }
    get() {
      return this.#total / this.#samples.length;
    }
  }
  
  
  // Add this class inside GPUCoordinator.svelte.js before the GPUCoordinator class
  class ScaleIndexManager {
    #scaleIds = [];
    #scaleIdToIndex = new Map();
  
    rebuild(allCurrentScaleIds) {
      console.log('Rebuilding scale indices:', allCurrentScaleIds);
      this.#scaleIds = [...allCurrentScaleIds];
      this.#scaleIdToIndex.clear();
      
      this.#scaleIds.forEach((scaleId, index) => {
        this.#scaleIdToIndex.set(scaleId, index);
      });
      
      console.log('Scale index mapping:', Object.fromEntries(this.#scaleIdToIndex));
      return this.#scaleIds.length;
    }
  
    getIndex(scaleId) {
      return this.#scaleIdToIndex.get(scaleId);
    }
  
    getCount() {
      return this.#scaleIds.length;
    }
  
    getAllScaleIds() {
      return [...this.#scaleIds];
    }
  }

    async initialize() {
      if (this.#initialized) return;
      try {
        await this.#initWebGPU();
        await this.#createSharedResources();
        await this.#initializeScaleBindGroupLayouts();
        this.#initialized = true;

      } catch (error) {
        console.error('GPUCoordinator initialization failed:', error);
        throw error;

    async #initializeScaleBindGroupLayouts() {
      // Create global SSBO layout (one layout for all scale types)
      const globalScaleLayout = this.#device.createBindGroupLayout({
        label: 'global-scale-ssbo-layout',
        entries: [{
          binding: 0,
          visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
          buffer: { 
            type: 'read-only-storage' // Changed from 'uniform' to 'read-only-storage'
          }
        }]
      });
      
      this.#globalScaleLayout = globalScaleLayout;
      this.#createGlobalScaleSSBO();
  
      for (let slot = 0; slot < this.#SSBO_SLOT_COUNT; slot++) {
        const layout = this.#globalScaleLayout;
        const slotOffset = slot * this.#slotByteSize;
        const bg = this.#device.createBindGroup({
          label: `global-scale-bind-group-slot-${slot}`,
          layout,
          entries: [{ binding: 0, resource: { buffer: this.#globalScaleSSBO, offset: slotOffset, size: this.#slotByteSize } }]
        });
        this.#globalScaleBindGroups[slot] = bg;
      }
      
      console.log('Initialized global SSBO layout, buffers, and bind group');
    }
  
    /**
     * Get or create the shared frame layout for all charts
     * @private
     */
    #getOrCreateFrameLayout() {
      if (!this.#frameLayout) {
        this.#frameLayout = this.#device.createBindGroupLayout({
          label: 'shared-frame-layout',
          entries: [{
            binding: 0,
            visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
            buffer: { type: 'uniform' }
          }]
        });
        console.log('Created shared frame layout for per-chart uniforms');
      }
      return this.#frameLayout;
    }
  
    #createGlobalScaleSSBO() {
      const scaleParamSize = 16; // 4 floats * 4 bytes 
      this.#slotByteSize = this.#maxScaleSlots * scaleParamSize;
      const totalSSBOSize = this.#slotByteSize * this.#SSBO_SLOT_COUNT;
      
      // Single SSBO with 3 slots
      this.#globalScaleSSBO = this.#device.createBuffer({
        size: totalSSBOSize,
        usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
        label: 'global-scale-ssbo-triple-buffered'
      });
      
      // Create staging buffer pool (pre-mapped)
      for (let i = 0; i < 4; i++) { // 4 staging buffers for safety
        const staging = this.#device.createBuffer({
          size: this.#slotByteSize,
          usage: GPUBufferUsage.MAP_WRITE | GPUBufferUsage.COPY_SRC,
          mappedAtCreation: true,
          label: `scale-staging-pool-${i}`
        });
        this.#stagingPool.push(staging);
      }
    }
  
    #updateScaleSSBOWithStaging() {
      if (this.#animatedScales.size === 0) return;
      
      // Get pre-mapped staging buffer from pool
      const staging = this.#stagingPool.pop();
      if (!staging) {
        console.warn('No staging buffer available in pool');
        return;
      }
      
      // Calculate destination slot  
      const slot = this.#frameId % this.#SSBO_SLOT_COUNT;
      const dstOffset = slot * this.#slotByteSize;
      
      // Write all scales synchronously to mapped staging
      const view = new Float32Array(staging.getMappedRange());
      this.#fillAllScales(view);
      // console.log(view[0], view[1], view[2])
      
      return { staging, dstOffset, slot };
    }
  
    #fillAllScales(view) {
      view.fill(0);
      const ids = this.#scaleIndexManager.getAllScaleIds();
      for (const id of ids) {
        const idx = this.#scaleIndexManager.getIndex(id);
        const scale = this.#scaleIdToInstance.get(id);
        if (idx === undefined || !scale) continue;
        const s = scale.gpuState;
        if (!s) continue;
        const off = idx * 4;
        view[off + 0] = s.domainMin;
        view[off + 1] = s.domainMax;
        view[off + 2] = s.domainScale || (1.0 / (s.domainMax - s.domainMin));
        view[off + 3] = 0;
      }
    }
  
  
    registerScale(scaleId, scaleType = 'linear') {
      if (!this.#initialized) {

    unregisterScale(scaleId) {

    #handleScaleTopologyChange() {
      // Build list of scale ids and a map id -> instance
      const ids = [];
      this.#scaleIdToInstance.clear();
  
      if (this.#scaleRegistry?.scaleUsage) {
        for (const scale of this.#scaleRegistry.scaleUsage.keys()) {
          if (scale?.id) {
            ids.push(scale.id);
            this.#scaleIdToInstance.set(scale.id, scale)
          }
        }
      }
  
      const count = this.#scaleIndexManager.rebuild(ids);
  
      // Grow SSBO if needed
      if (count > this.#maxScaleSlots) {
        let newSlots = this.#maxScaleSlots;
        while (newSlots < count ) newSlots *= 2;
        this.#maxScaleSlots = newSlots;
        this.#recreateGlobalScalesSSBOAndBindGroups();
      }
  
      // Indices chnaged -> per-chart uniforms must be rewritten
      for (const [chartid] of this.#registeredCharts) {
        this.updateChartFrameUniforms(chartid);
      }
  
      // Force one full upload so SSBO has initial values
      this.#ssboNeedsUpload = true
    }
  
    #recreateGlobalScalesSSBOAndBindGroups() {
      try { this.#globalScaleSSBO?.destroy?.(); } catch {}
      // Drop old staging
      for (const s of this.#stagingPool) {
        try { s.unmap(); } catch {}
        try { s.destroy?.(); } catch {}
      }
      this.#stagingPool.length = 0;
  
      // Recreate SSBO & staging and the 3 bind groups
      this.#createGlobalScaleSSBO();
      this.#globalScaleBindGroups.length = 0;
      for ( let slot = 0; slot < this.#SSBO_SLOT_COUNT; slot++) {
        const layout = this.#globalScaleLayout;
        
        const offset = slot * this.#slotByteSize;
        this.#globalScaleBindGroups[slot] = this.#device.createBindGroup({
          layout,
          entries: [{ binding: 0, resource: { buffer: this.#globalScaleSSBO, offset, size: this.#slotByteSize } }]
        });
      }
    }
  
    /**
     * Create frame uniform buffer and bind group for a specific chart
     * @private
     */
    #createChartFrameResources(chartId) {
      const chartConfig = this.#registeredCharts.get(chartId);
      if (!chartConfig || chartConfig.frameUniformsBuffer) {
        return; // Chart doesn't exist or resources already created
      }
  
      try {
        chartConfig.frameUniformsBuffer = this.#device.createBuffer({
          size: 48, // 10 x f32 = 40 bytes 
          usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST,
          label: `frame-uniforms-${chartId}`
        });
  
        // Create the bind group
        this.#createChartFrameBindGroup(chartId);
        
        console.log(`Created frame resources for chart ${chartId}`);
        
      } catch (error) {
        console.error(`Failed to create frame resources for chart ${chartId}:`, error);
      }
    }
  
    /**
     * Create frame bind group for a specific chart
     * @private
     */
    #createChartFrameBindGroup(chartId) {
      const chartConfig = this.#registeredCharts.get(chartId);
      if (!chartConfig?.frameUniformsBuffer || chartConfig.perFrameBindGroup) {
        return; // Buffer not ready or bind group already exists
      }
  
      try {
        const frameLayout = this.#getOrCreateFrameLayout();
        
        chartConfig.perFrameBindGroup = this.#device.createBindGroup({
          label: `per-frame-bind-group-${chartId}`,
          layout: frameLayout,
          entries: [{
            binding: 0,
            resource: { buffer: chartConfig.frameUniformsBuffer }
          }]
        });
        
        console.log(`Created frame bind group for chart ${chartId}`);
        
      } catch (error) {
        console.error(`Failed to create frame bind group for chart ${chartId}:`, error);
      }
    }
  
    /**
     * Update frame uniforms for a specific chart
     * Called when canvas dimensions or DPR changes
     */
    updateChartFrameUniforms(chartId) {
      const chartConfig = this.#registeredCharts.get(chartId);
      if (!chartConfig?.contextConfigured) {
        console.debug(`Queueing frame uniforms update for unregistered chart ${chartId}`);
        this.#pendingFrameUpdates.set(chartId, { timestamp: Date.now() });
        return;
      }
  
      try {
        // Create resources if they don't exist yet
        if (!chartConfig.frameUniformsBuffer) {
          this.#createChartFrameResources(chartId);
        }
  
        const dpr = window.devicePixelRatio || 1;
        const canvas = chartConfig.canvas;
  
        // Compute in logical CSS px, then convert ranges to physical px
        const resX = canvas.width; 
        const resY = canvas.height;
        
        const rangeMinX = 0 * dpr;
        const rangeMinY = canvas.clientHeight * dpr;
        const rangeMaxX = canvas.clientWidth * dpr;
        const rangeMaxY = 0 * dpr;
  
        const xScaleIndex = this.#scaleIndexManager.getIndex(chartConfig.xScaleId) ?? 0;
        const yScaleIndex = this.#scaleIndexManager.getIndex(chartConfig.yScaleId) ?? 1;
        
        // Build 48-byte uniform payload
        const buf = new ArrayBuffer(48);
        const f32 = new Float32Array(buf);
        const u32 = new Uint32Array(buf);
  
        // Floats: resolution(2), rangeMin(2), rangeMax(2), devicePixelRation, _pad0
        f32[0] = resX;
        f32[1] = resY;
        f32[2] = rangeMinX;
        f32[3] = rangeMinY;
        f32[4] = rangeMaxX;
        f32[5] = rangeMaxY;
        f32[6] = dpr;
        f32[7] = 0.0;
  
        // U32s: xScaleIndex, yScaleIndex, _pad1 (2x)
        u32[8] = xScaleIndex >>> 0;
        u32[9] = yScaleIndex >>> 0;
        u32[10] = 0;
        u32[11] = 0;
        
        this.#device.queue.writeBuffer(chartConfig.frameUniformsBuffer, 0, buf);
        
      } catch (error) {
        console.error(`Failed to update frame uniforms for chart ${chartId}:`, error);
      }
    }
  
    // Process pending updates when chart becomes ready
    #processPendingFrameUpdates(chartId) {
      const pendingUpdate = this.#pendingFrameUpdates.get(chartId);
      if (pendingUpdate) {
        console.debug(`Processing queued frame uniforms update for chart ${chartId}`);
        this.updateChartFrameUniforms(chartId);
        this.#pendingFrameUpdates.delete(chartId);
      }
    }

    #coordinatedRender(chartsNeedingRender) {
  
      // await this.#device.queue.onSubmittedWorkDone()
      // await new Promise(resolve => setTimeout(resolve, 0));

      //   await this.#inflightPromises[0];
      //   this.#inflightPromises.shift();
      // }
  
      try {
  
        let stagingResult = null;
  
        // Update scales with staging buffer
        if (this.#ssboNeedsUpload || this.#animatedScales.size > 0) {
          stagingResult = this.#updateScaleSSBOWithStaging();
          this.#ssboNeedsUpload = false
        }
  
        const commandEncoder = this.#device.createCommandEncoder({ label: 'frame-encoder' });
  
        // Copy staging to SSBO if we have scale updates
        if (stagingResult) {
          commandEncoder.copyBufferToBuffer(
            stagingResult.staging, 0,
            this.#globalScaleSSBO, stagingResult.dstOffset,
            this.#slotByteSize
          );
        }
  
        const currentSlot = this.#frameId % this.#SSBO_SLOT_COUNT;
        const currentScaleBindGroup = this.#globalScaleBindGroups[currentSlot]
  
        for (const chartConfig of chartsNeedingRender) {
  
          if (chartConfig.needsPlanRebuild) {
            this.#buildRenderPlanForChart(chartConfig);
          }
          
          const passEncoder = commandEncoder.beginRenderPass(
            this.#createRenderPassDescriptor(
              chartConfig.context.getCurrentTexture().createView(),
              chartConfig.canvas,
              chartConfig.chartId
            )
          );
          
          // Get the per-frame bind group from the chart (shared by all layers)
          const chartPerFrameBindGroup = chartConfig.perFrameBindGroup;
          if (!chartPerFrameBindGroup) {
            console.debug(`Skipping render for chart ${chartConfig.chartId} - chart frame bind group not ready`);
            passEncoder.end();
            continue; // Skip to next chart
          }
  
          // Set the chart's frame bind group once (all layers in this chart will use it)
          passEncoder.setBindGroup(0, chartPerFrameBindGroup);
          passEncoder.setBindGroup(2, currentScaleBindGroup);
  
          let currentPipeline = null;
          let currentPerDataBindGroup = null;
  
          for (const layer of chartConfig.renderPlan) {
            const renderer = layer.renderer;
            
            // NEW: Only check for data bind groups (frame bind group is now chart-level)
            if (!renderer || !renderer.perDataBindGroup) {
              console.debug(`Skipping layer ${layer.id} - renderer data bind group not ready`);
              continue;
            }
  
            const pipelineToUse = layer.cachedPipeline;
            const perDataBindGroupToUse = renderer.perDataBindGroup;
  
            // Set pipeline only if different
            if (pipelineToUse !== currentPipeline) {
              passEncoder.setPipeline(pipelineToUse);
              currentPipeline = pipelineToUse;
            }
            
            if (perDataBindGroupToUse !== currentPerDataBindGroup) {
              passEncoder.setBindGroup(1, perDataBindGroupToUse);
              currentPerDataBindGroup = perDataBindGroupToUse;
            }
  
            layer.draw(passEncoder);
          }
          passEncoder.end();
  
        }
  
        if (this.#canTimestamp) {
          commandEncoder.resolveQuerySet(this.#querySet, 0, this.#querySet.count, this.#resolveBuffer, 0);
          if (this.#resultBuffer.mapState === 'unmapped') {
            commandEncoder.copyBufferToBuffer(this.#resolveBuffer, 0, this.#resultBuffer, 0, this.#resultBuffer.size);
          }
        }
  
        // Unmap staging before submit
        if (stagingResult) {
          stagingResult.staging.unmap();
        }
  
        this.#device.queue.submit([commandEncoder.finish()]);
  
        const p = this.#device.queue.onSubmittedWorkDone();
        this.#inflightPromises.push(p);
  
        // Remove when resolved
        p.then(() => {
          // console.log(`Frame presented at: ${performance.now()}`);
          const index = this.#inflightPromises.indexOf(p);
          if (index !== -1) {
            this.#inflightPromises.splice(index, 1);
          }
        });
  
        if (this.#canTimestamp && this.#resultBuffer.mapState === 'unmapped') {
          this.#resultBuffer.mapAsync(GPUMapMode.READ).then(() => {
            const times = new BigInt64Array(this.#resultBuffer.getMappedRange());
            const gpuTime = Number(times[1] - times[0]);
            this.#gpuAverage.addSample(gpuTime / 1000);
            this.timings.gpu = this.#gpuAverage.get().toFixed(2);
            this.#resultBuffer.unmap();
          });
        }
  
        if (stagingResult) {
          stagingResult.staging.mapAsync(GPUMapMode.WRITE).then(() => {
              this.#stagingPool.push(stagingResult.staging);
          });
        }
  
  
        // // Async remap staging back to pool
        // if (stagingResult) {
        //   const p = this.#device.queue.onSubmittedWorkDone().then(() => {
        //     return stagingResult.staging.mapAsync(GPUMapMode.WRITE).then(() => {
        //       this.#stagingPool.push(stagingResult.staging);
        //     });
        //   });
        //   this.#inflightPromises.push(p);
        // }
        
        this.#frameId++;
        
        
      } catch (error) {
        console.error('Coordinated render failed:', error);
        this.#fallbackRender(chartsNeedingRender);
      }
    }
  
    /**
     * Fallback rendering when coordinated render fails
     * @private
     */
    #fallbackRender(chartsNeedingRender) {
      console.warn('Using fallback render path');
      for (const chartConfig of chartsNeedingRender) {
        try {
          const commandEncoder = this.#device.createCommandEncoder();
          const passEncoder = commandEncoder.beginRenderPass(
            this.#createRenderPassDescriptor(
              chartConfig.context.getCurrentTexture().createView(),
              chartConfig.canvas,
              chartConfig.chartId
            )
          );
  
          const chartPerFrameBindGroup = chartConfig.perFrameBindGroup;
          if (!chartPerFrameBindGroup) {
            passEncoder.end();
            continue;
          }
  
          for (const layer of chartConfig.layers.values()) {
            const renderer = layer.renderer;
            if (renderer?.pipeline && renderer?.perDataBindGroup) {
              const currentSlot = this.#frameId % this.#SSBO_SLOT_COUNT;
              const currentScaleBG = this.#globalScaleBindGroups[currentSlot];
              passEncoder.setPipeline(renderer.pipeline);
              passEncoder.setBindGroup(0, chartPerFrameBindGroup); 
              passEncoder.setBindGroup(1, renderer.perDataBindGroup);
              passEncoder.setBindGroup(2, currentScaleBG);
              layer.draw(passEncoder);
            }
          }
          passEncoder.end();
          this.#device.queue.submit([commandEncoder.finish()]);
  
          this.#frameId++;
  
        } catch (error) {
          console.error(`Failed to render chart ${chartConfig.chartId}:`, error);
        }
      }
    }

    /**
     * Update the scale ID mapping for a specific chart
     * Called by ChartScales when scale instances are resolved or change
     */
    setChartScaleMapping(chartId, { xScaleId, yScaleId }) {
      const chartConfig = this.#registeredCharts.get(chartId);
      if (!chartConfig) {
        // Queue the mapping for when chart is registered
        console.debug(`Queueing scale mapping for unregistered chart ${chartId}`);
        this.#pendingScaleMappings.set(chartId, { xScaleId, yScaleId });
        return;
      }
  
      const changed = chartConfig.xScaleId !== xScaleId || chartConfig.yScaleId !== yScaleId;
      
      chartConfig.xScaleId = xScaleId;
      chartConfig.yScaleId = yScaleId;
      
      if (changed) {
        console.debug(`Updated scale mapping for chart ${chartId}:`, { xScaleId, yScaleId });
        
        // Trigger rebuild of render plan since scale bindings may have changed
        chartConfig.needsPlanRebuild = true;
        this.invalidateChart(chartId);
      }
  }
  
    invalidateChart(chartId) {
      if (this.#registeredCharts.has(chartId)) {
        this.#dirtyChartIDs.add(chartId);
        this.#ensureLoopIsActive();
      }
    }
  
    addAnimatingScale(scale) {
      let idx = this.#scaleIndexManager.getIndex(scale.id);
      if (idx === undefined) {
        this.registerScale(scale.id, 'linear');
        idx = this.#scaleIndexManager.getIndex(scale.id);
        if (idx === undefined) {
          console.error(`Scale ${scale.id} still not registered after auto-register`);
          return;
        }
      }
      
      this.#animatedScales.add(scale);
      this.#ensureLoopIsActive();
    }
  
    removeAnimatingScale(scale) {
      this.#animatedScales.delete(scale);
    }
  
    #ensureLoopIsActive() {
      if (this.#animationRafId) return;
  
      if (this.#animatedScales.size > 0 && !this.#animationActive) {
        this.#animationActive = true;
        this.#accumulator = 0;
        this.#lastFrameTime = 0;
      }
  
      this.#animationRafId = requestAnimationFrame(this.#animationLoop);
    }
  
    registerChart(chartId, canvas, config) {
      if (this.#registeredCharts.has(chartId)) {
        console.warn(`Chart ${chartId} already registered, updating config`);
      }
  
      const chartConfig = {
        chartId,
        canvas,
        zIndex: config.zIndex || 0,
        visible: true,
        context: canvas.getContext('webgpu'),
        contextConfigured: false,
        layers: new Map(),
        renderPlan: [],
        needsPlanRebuild: true,
        xScaleId: null,
        yScaleId: null,
        frameUniformsBuffer: null,
        perFrameBindGroup: null,
      };
  
      this.#registeredCharts.set(chartId, chartConfig);
  
      // Process any pending scale mappings for this chart
      const pendingMapping = this.#pendingScaleMappings.get(chartId);
      if (pendingMapping) {
        console.debug(`Applying queued scale mapping for chart ${chartId}:`, pendingMapping);
        this.setChartScaleMapping(chartId, pendingMapping);
        this.#pendingScaleMappings.delete(chartId);
      }
  
      return {
        setContextReady: () => {
          const chart = this.#registeredCharts.get(chartId);
          if (chart) {
            chart.contextConfigured = true;
            this.#processPendingFrameUpdates(chartId);
          }
        },
        setVisible: (visible) => {
          const config = this.#registeredCharts.get(chartId);
          if (config && config.visible !== visible) {
            config.visible = visible;
            this.invalidateChart(chartId);
          }
        },
        invalidate: () => {
          this.invalidateChart(chartId);
        },
        remove: () => {
          this.unregisterChart(chartId);
        }
      };
    }
  
    unregisterChart(chartId) {
      const chartConfig = this.#registeredCharts.get(chartId);
      if (chartConfig) {
        // NEW: Clean up per-chart frame resources
        if (chartConfig.frameUniformsBuffer && !chartConfig.frameUniformsBuffer.destroyed) {
          chartConfig.frameUniformsBuffer.destroy();
        }
        // Note: perFrameBindGroup doesn't need explicit cleanup
  
        this.#registeredCharts.delete(chartId);
  
        // Existing MSAA cleanup
        const msaa = this.#msaaCache.get(chartId);
        if (msaa) {
          msaa.texture.destroy();
          this.#msaaCache.delete(chartId);
          console.debug(`Destroyed cached MSAA resources for chart: ${chartId}`);
        }
  
        // Existing pending mappings cleanup
        this.#pendingScaleMappings.delete(chartId);
  
        console.log(`Unregistered chart: ${chartId}`);
      }
    }
  
    registerLayer(chartId, layerConfig) {
      const chart = this.#registeredCharts.get(chartId);
      if (!chart) {
        console.error(`Attempted to register layer on non-existent chart: ${chartId}`);
        return { remove: () => {} };
      }
  
      const layerId = layerConfig.id || `layer-${crypto.randomUUID().slice(0, 8)}`;
      chart.layers.set(layerId, { ...layerConfig, id: layerId, zIndex: layerConfig.zIndex ?? 0 });
  
      chart.needsPlanRebuild = true;
      console.log(`📋 Layer registered: chart ${chartId} now has ${chart.layers.size} layers, needsPlanRebuild=${chart.needsPlanRebuild}`);
      this.invalidateChart(chartId);
  
      return {
        remove: () => this.unregisterLayer(chartId, layerId)
      };
    }
  
    unregisterLayer(chartId, layerId) {
      const chart = this.#registeredCharts.get(chartId);
      if (chart?.layers.has(layerId)) {
        chart.layers.delete(layerId);
        chart.needsPlanRebuild = true;
        this.invalidateChart(chartId);
      }
    }
  
    #buildRenderPlanForChart(chartConfig) {
      // console.log(`🏗️ Building render plan for chart ${chartConfig.chartId}: ${chartConfig.layers.size} layers available`);
    
      const layers = Array.from(chartConfig.layers.values());
      if (layers.length === 0) {
        console.log(`⚠️ No layers to render for chart ${chartConfig.chartId}`);
        chartConfig.renderPlan = [];
        chartConfig.needsPlanRebuild = false;
        return;
      }
  
      const zIndexGroups = new Map();
      for (const layer of layers) {
        const zIndex = layer.zIndex;
        if (!zIndexGroups.has(zIndex)) zIndexGroups.set(zIndex, []);
        zIndexGroups.get(zIndex).push(layer);
      }
  
      const sortedZLevels = Array.from(zIndexGroups.keys()).sort((a, b) => a - b);
      const newPlan = [];
      for (const zLevel of sortedZLevels) {
        const layersInZLevel = zIndexGroups.get(zLevel);
        const markTypeGroups = new Map();
        for (const layer of layersInZLevel) {
          const markType = layer.markType || 'unknown';
          if (!markTypeGroups.has(markType)) markTypeGroups.set(markType, []);
          markTypeGroups.get(markType).push(layer);
        }
        for (const batchedLayers of markTypeGroups.values()) {
          newPlan.push(...batchedLayers);
        }
      }
  
      // Enrich the plan with cached references
      for (const layer of newPlan) {
        layer.cachedPipeline = layer.renderer?.pipeline;
        layer.cachedPerFrameBindGroup = layer.renderer?.perFrameBindGroup;
        layer.cachedPerDataBindGroup = layer.renderer?.perDataBindGroup;
        layer.isDualBindGroupReady = !!(layer.cachedPerFrameBindGroup && layer.cachedPerDataBindGroup);
      }
  
      chartConfig.renderPlan = newPlan;
      chartConfig.needsPlanRebuild = false;
      // console.log(`✅ Render plan built for chart ${chartConfig.chartId}: ${newPlan.length} layers in plan`);
    }
  
    #createRenderPassDescriptor(textureView, canvas, chartId) {
      if (this.#msaaSampleCount > 1) {
        let msaa = this.#msaaCache.get(chartId);
  
        if (!msaa || 
            msaa.texture.width !== canvas.width ||
            msaa.texture.height !== canvas.height) {
            
          msaa?.texture.destroy();
          
          const newTexture = this.#device.createTexture({
            size: { width: canvas.width, height: canvas.height },
            sampleCount: this.#msaaSampleCount,
            format: this.#presentationFormat,
            usage: GPUTextureUsage.RENDER_ATTACHMENT
          });
  
          msaa = {
            texture: newTexture,
            view: newTexture.createView() 
          };
  
          this.#msaaCache.set(chartId, msaa);
        }
  
        return {
          colorAttachments: [{
            view: msaa.view, 
            resolveTarget: textureView,
            clearValue: { r: 0.0, g: 0.0, b: 0.0, a: 0.0 },
            loadOp: 'clear',
            storeOp: 'store',
          }],
          ...(this.#canTimestamp && {
            timestampWrites: {
              querySet: this.#querySet,
              beginningOfPassWriteIndex: 0,
              endOfPassWriteIndex: 1,
            },
          }),
        };
      } else {
        return {
          colorAttachments: [{
            view: textureView,
            clearValue: { r: 0.0, g: 0.0, b: 0.0, a: 0.0 },
            loadOp: 'clear',
            storeOp: 'store',
          }],
          ...(this.#canTimestamp && {
            timestampWrites: {
              querySet: this.#querySet,
              beginningOfPassWriteIndex: 0,
              endOfPassWriteIndex: 1,
            },
          }),
        };
      }
    }
  
    #getOrCreateDataLayout(markType, dataBindingConfig) {
      const layoutKey = `${markType}-${JSON.stringify(dataBindingConfig)}`;
      
      if (!this.#dataLayouts.has(layoutKey)) {
        const layout = this.#device.createBindGroupLayout({
          label: `${markType}-data-layout`,
          entries: dataBindingConfig.map(config => ({
            binding: config.binding,
            visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
            buffer: { type: config.type || 'read-only-storage' }
          }))
        });
        this.#dataLayouts.set(layoutKey, layout);
      }
      
      return this.#dataLayouts.get(layoutKey);
    }
  
    async createPipelineForMarkType(markType, scaleTypes, markShaderCode, vertexBufferLayout, dataBindingConfig) {
      console.log(`Creating pipeline for ${markType} with scales: ${scaleTypes.x}-${scaleTypes.y}`);
  
      // Generate shader using template system
      const templateResult = scaleTemplateSystem.generateShader(scaleTypes, markShaderCode);
      
      // Get or create stable layouts
      const frameLayout = this.#getOrCreateFrameLayout();
      const dataLayout = this.#getOrCreateDataLayout(markType, dataBindingConfig);
      const globalScaleLayout = this.#globalScaleLayout;;
  
  
      if (!globalScaleLayout) {
        throw new Error('Global scale SSBO layout not found - ensure SSBO system is initialized');
      }
  
      // Create pipeline layout
      const pipelineLayout = this.#device.createPipelineLayout({
        label: `${markType}-pipeline-layout`,
        bindGroupLayouts: [frameLayout, dataLayout, globalScaleLayout]
      });
  
      // Create or get shared shader
      console.log('🏭 Creating shader module...');
      const shaderModule = await this.getOrCreateSharedShader(markType, templateResult.shaderCode);
      console.log('🏭 Shader module created successfully');
  
      // Create pipeline
      console.log('🏭 Creating render pipeline...');
      const pipeline = await this.getOrCreateSharedPipeline(
        `${markType}-${scaleTypes.x}-${scaleTypes.y}`, // Include scale types in cache key
        async () => {
          const msaaSampleCount = this.#msaaSampleCount;
  
          console.log('🏭 Pipeline config:', {
            msaa: msaaSampleCount,
            format: this.#presentationFormat,
            vertexBuffers: vertexBufferLayout?.length || 0
          });
          
          return this.#device.createRenderPipeline({
            label: `${markType}-${scaleTypes.x}-${scaleTypes.y}-pipeline`,
            layout: pipelineLayout,
            vertex: {
              module: shaderModule,
              entryPoint: 'vertexMain',
              buffers: vertexBufferLayout || [],
            },
            fragment: {
              module: shaderModule,
              entryPoint: 'fragmentMain',
              targets: [{
                format: this.#presentationFormat,
                blend: {
                  color: {
                    srcFactor: 'src-alpha',
                    dstFactor: 'one-minus-src-alpha',
                    operation: 'add',
                  },
                  alpha: {
                    srcFactor: 'one',
                    dstFactor: 'one-minus-src-alpha',
                    operation: 'add',
                  },
                },
              }],
            },
            primitive: { topology: 'triangle-list' },
            multisample: { count: msaaSampleCount > 1 ? msaaSampleCount : 1 },
          });
        }
      );
  
       console.log('🏭 Pipeline created successfully');
  
      return {
        pipeline,
        frameLayout,
        dataLayout
      };
    }
  
    // ===== SHARED RESOURCE METHODS WITH REFERENCE COUNTING =====
  
    async getOrCreateSharedPipeline(markType, createFn) {
      console.log(`Pipeline request for ${markType}, existing:`, !!this.#sharedResources.pipelines.get(markType));
    
      const existing = this.#sharedResources.pipelines.get(markType);
      
      if (existing) {
        existing.refCount++;
        return existing.pipeline;
      }
  
      // Check if already creating to prevent race conditions
      if (this.#creatingPipelines.has(markType)) {
        // Wait for the other creation to complete
        while (this.#creatingPipelines.has(markType)) {
          await new Promise(resolve => setTimeout(resolve, 10));
        }
        // Now get the created pipeline
        const newExisting = this.#sharedResources.pipelines.get(markType);
        if (newExisting) {
          newExisting.refCount++;
          console.log(`Pipeline request for ${markType}, was reused after creation:`, !!this.#sharedResources.pipelines.get(markType));
          return newExisting.pipeline;
        }
      }
  
      // Mark as creating
      this.#creatingPipelines.add(markType);
  
      try {
        console.log(`Creating new pipeline for ${markType}`);
        const pipeline = await createFn();
        
        this.#sharedResources.pipelines.set(markType, {
          pipeline,
          refCount: 1,
          createdAt: Date.now()
        });
  
        return pipeline;
      } finally {
        this.#creatingPipelines.delete(markType);
      }
    }
  
    releaseSharedPipeline(markType) {
      const resource = this.#sharedResources.pipelines.get(markType);
      if (!resource) return;
  
      resource.refCount--;
  
      if (resource.refCount <= 0) {
        this.#sharedResources.pipelines.delete(markType);
        console.log(`Released shared pipeline: ${markType}`);
      }
    }
  
    async getOrCreateSharedShader(markType, shaderCode) {
      const existing = this.#sharedResources.shaderModules.get(markType);
      
      if (existing) {
        existing.refCount++;
        return existing.module;
      }
  
      const module = this.#device.createShaderModule({ code: shaderCode });
      
      this.#sharedResources.shaderModules.set(markType, {
        module,
        refCount: 1,
        createdAt: Date.now()
      });
  
      return module;
    }
  
    releaseSharedShader(markType) {
      const resource = this.#sharedResources.shaderModules.get(markType);
      if (!resource) return;
  
      resource.refCount--;
  
      if (resource.refCount <= 0) {
        this.#sharedResources.shaderModules.delete(markType);
        console.log(`Released shared shader: ${markType}`);
      }
    }
  
    getSharedUniformBuffer(name) {
      const resource = this.#sharedResources.uniformBuffers.get(name);
      if (resource) {
        resource.refCount++;
        return resource.buffer;
      }
      return null;
    }
  
    releaseSharedUniformBuffer(name) {
      const resource = this.#sharedResources.uniformBuffers.get(name);
      if (resource) {
        resource.refCount--;
        
        if (resource.refCount <= 0) {
          if (resource.buffer && !resource.buffer._isDestroyed) {
            resource.buffer.destroy();
          }
          this.#sharedResources.uniformBuffers.delete(name);
          console.log(`Released shared uniform buffer: ${name}`);
        }
      }
    }

    getSharedPipeline(markType) {
      return this.#sharedResources.pipelines.get(markType)?.pipeline;
    }
  
    getSharedShaderModule(markType) {
      return this.#sharedResources.shaderModules.get(markType)?.module;
    }

    // ===== INITIALIZATION METHODS =====
  
    async #initWebGPU() {
      if (typeof window === 'undefined' || typeof navigator === 'undefined') {
        throw new Error('WebGPU initialization attempted server-side');
      }
  
      if (!navigator.gpu) {
        throw new Error('WebGPU not supported');
      }
  
      const adapter = await navigator.gpu.requestAdapter({
        powerPreference: "high-performance"
      });
  
      if (!adapter) {
        throw new Error('No appropriate GPUAdapter found');
      }
  
      this.#canTimestamp = adapter.features.has('timestamp-query');
  
      this.#device = await adapter.requestDevice({
        requiredFeatures: [
          ...(this.#canTimestamp ? ['timestamp-query'] : []),
         ],
      });
  
      this.#presentationFormat = navigator.gpu.getPreferredCanvasFormat();
  
      // Timestamp querying resources
      if (!this.#canTimestamp) {
        console.warn('Timestamp queries not supported');
        return;
      }
      
      this.#querySet = this.device.createQuerySet({
        type: 'timestamp',
        count: 2,
      });
        
      this.#resolveBuffer = this.device.createBuffer({
        size: this.#querySet.count * 8,
        usage: GPUBufferUsage.QUERY_RESOLVE | GPUBufferUsage.COPY_SRC,
      });
        
      this.#resultBuffer = this.device.createBuffer({
        size: this.#resolveBuffer.size,
        usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ,
      });
    }
  
    async #createSharedResources() {
      const device = this.#device;
  
      const dprBuffer = device.createBuffer({
        size: 4,
        usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST,
      });
  
      this.#sharedResources.uniformBuffers.set('devicePixelRatio', {
        buffer: dprBuffer,
        refCount: 0,
        createdAt: Date.now()
      });
  
      this.#updateSharedUniforms();
    }
  
    #updateSharedUniforms() {
      if (!this.#device) return;
  
      const devicePixelRatio = window.devicePixelRatio || 1;
      const dprResource = this.#sharedResources.uniformBuffers.get('devicePixelRatio');
  
      if (dprResource?.buffer) {
        this.#device.queue.writeBuffer(dprResource.buffer, 0, new Float32Array([devicePixelRatio]));
      }
    }
  
    // ===== COMPREHENSIVE CLEANUP =====
  
    destroy() {
      console.log('Destroying GPUCoordinator with comprehensive cleanup');
      
      // Stop animation loop
      if (this.#animationRafId !== null) {
        cancelAnimationFrame(this.#animationRafId);
        this.#animationRafId = null;
      }
      
      this.#animationActive = false;
      this.#animatedScales.clear();
  
      // Cleanup MSAA textures
      for (const msaa of this.#msaaCache.values()) {
        if (msaa.texture && !msaa.texture.destroyed) {
          msaa.texture.destroy();
        }
      }
      this.#msaaCache.clear();
  
      // Cleanup shared resources with reference counting
      for (const [name, resource] of this.#sharedResources.uniformBuffers) {
        if (resource.buffer && !resource.buffer._isDestroyed) {
          resource.buffer.destroy();
        }
      }
      this.#sharedResources.uniformBuffers.clear();
      this.#sharedResources.pipelines.clear();
      this.#sharedResources.shaderModules.clear();
      this.#sharedResources.bindGroupLayouts.clear();
  
      // Cleanup bind group layouts
      // this.#scaleBindGroupLayouts.clear();
      this.#globalScaleLayout = null;
  
      // Cleanup charts and layers
      this.#registeredCharts.clear();
      this.#dirtyChartIDs.clear();
  
      // Cleanup performance tracking
      this.#frameHistory = [];
      this.#frameTimings = [];
  
      // Cleanup race condition protection
      this.#creatingPipelines.clear();
  
      // Clean up layouts
      this.#dataLayouts.clear();
      this.#frameLayout = null;
  
      this.#pendingScaleMappings.clear();
  
      // Destroy device last
      if (this.#device) {
        this.#device.destroy();
        this.#device = null;
      }
  
      this.#initialized = false;
      console.log('GPUCoordinator destruction complete');
    }
  }
  
  // Context functions remain unchanged...

      let newMin = initialDomain[0] + dataDelta;
      let newMax = initialDomain[1] + dataDelta;
      
      // Apply constraints
      const range = newMax - newMin;
      if (newMin < this.domainLimits.min) {
        newMin = this.domainLimits.min;
        newMax = newMin + range;
      } else if (newMax > this.domainLimits.max) {
        newMax = this.domainLimits.max;

          newMin = this.domainLimits.min;

    _applyInput(now) {
      let inputApplied = false;
  
      //  Handle zoom accumulation
      if (this.#frameAccumulator.hasZoomUpdates) {
        this.#updateSpringConfig(springConfigs.zoom);
        this.#calculateZoomedDomain(
          this.#target,
          this.#frameAccumulator.zoomCenter,
          Math.exp(this.#frameAccumulator.logZoomDelta)
        );
        // this.#smoothedTarget[0] = this.#target[0];
        // this.#smoothedTarget[1] = this.#target[1];
        inputApplied = true;
      }
  
      // --- Handle Pan Velocity Calculation ---
      if (this.#activePanContext && this.#inputRingBuffer.length > 1) {
        const sampleTime = now - this.#jitterBufferMs;
        const resampledPoint = this.#resampleAt(sampleTime);
        
        if (resampledPoint) {
          const { screenToDataRatio } = this.#activePanContext;
          
          const screenVelocity = this.#activePanContext.activePanX
            ? resampledPoint.velocityX
            : resampledPoint.velocityY;
          
          const dataVelocity = -screenVelocity * screenToDataRatio * 1000; // data units per second
          
          // Smooth the clean, resampled velocity.
          this.#panVelocity += (dataVelocity - this.#panVelocity) * this.#panVelocitySmoothingFactor;
        }
      } else if (this.#activePanContext) {
        // If there's no new input, the velocity should decay.
        this.#panVelocity *= 0.9; // Decay factor
      }
  
      // Reset zoom accumulator
      this.#frameAccumulator.logZoomDelta = 0;
      this.#frameAccumulator.zoomCenter = 0;
      this.#frameAccumulator.hasZoomUpdates = false;
  
      if (inputApplied) {
        this.#ensureAnimating();
      }
    }

    _updatePhysics(deltaTime) {
      const dt = deltaTime / 1000.0;
      const { min: minLimit, max: maxLimit } = this.domainLimits;
  
      // --- 1. UPDATE TARGET (if panning) ---
      if (this.#activePanContext) {
        // The target is driven by our smoothed, resampled velocity.
        const panDelta = this.#panVelocity * dt;
        this.#target[0] += panDelta;
        this.#target[1] += panDelta;
      }
      
      // --- 2. CALCULATE FORCES (for the spring) ---
      this.#physicsPrevious[0] = this.#physicsCurrent[0];
      this.#physicsPrevious[1] = this.#physicsCurrent[1];
      
      // Calculate distance for the main spring force
      let distance = this.#target[0] - this.#physicsCurrent[0];
      
      // Calculate overpan for the boundary resistance force
      const domainWidth = this.#physicsCurrent[1] - this.#physicsCurrent[0];
      let overpan = 0;
      if (this.#physicsCurrent[0] < minLimit) {
        overpan = minLimit - this.#physicsCurrent[0];
      } else if (this.#physicsCurrent[0] + domainWidth > maxLimit) {
        overpan = maxLimit - (this.#physicsCurrent[0] + domainWidth);
      }
      
      // --- 3. APPLY FORCES & INTEGRATE (the core physics step) ---
      
      // Main spring force (always active, pulls towards target)
      const springForce = distance * this.#currentSpringConfig.stiffness;
      
      // Boundary resistance force (only active when overpanning)
      const resistanceForce = overpan * this.#currentSpringConfig.stiffness * this.#boundaryResistanceFactor;
      
      // Damping force (always active, opposes motion)
      const dampingForce = -this.#velocity[0] * this.#currentSpringConfig.damping;
      
      // Sum forces to get acceleration
      const acceleration = springForce + dampingForce + resistanceForce;
      
      // Integrate acceleration to get new velocity
      this.#velocity[0] += acceleration * dt;
      
      // Integrate velocity to get new position
      this.#physicsCurrent[0] += this.#velocity[0] * dt;
      
      // The other side of the domain follows rigidly
      this.#physicsCurrent[1] = this.#physicsCurrent[0] + domainWidth;
      this.#velocity[1] = this.#velocity[0];
  
      // --- 4. HANDLE MOMENTUM (after user releases) ---
      if (!this.#activePanContext) {
        // If coasting, apply momentum decay directly to the velocity
        this.#velocity[0] *= this.#panMomentumFactor;
        this.#velocity[1] *= this.#panMomentumFactor;
  
        // Boundary collision check for momentum
        if ((this.#physicsCurrent[0] < minLimit && this.#velocity[0] < 0) || (this.#physicsCurrent[1] > maxLimit && this.#velocity[0] > 0)) {
            this.#velocity[0] *= -this.#momentumBoundaryDamping; // Damp and reverse
            this.#velocity[1] *= -this.#momentumBoundaryDamping;
        }
      }
  
      // --- 5. CHECK SETTLING CONDITION ---
      const isSettled = Math.abs(this.#velocity[0]) < 0.001 && Math.abs(this.#target[0] - this.#physicsCurrent[0]) < 0.001;
  
      if (isSettled && !this.#activePanContext) {
        this.#velocity = [0, 0];
        this.#isAnimating = false;
        this.#gpuCoordinator.removeAnimatingScale(this);
      }
    }

    #applyBoundaryResistance(dt) {
      if (!this.#activePanContext) return;
      
      const { min: minLimit, max: maxLimit } = this.domainLimits;
      const domainWidth = this.#target[1] - this.#target[0];
      let overpan = 0;
  
      if (this.#target[0] < minLimit) {
        overpan = minLimit - this.#target[0];
      } else if (this.#target[0] + domainWidth > maxLimit) {
        overpan = maxLimit - (this.#target[0] + domainWidth);
      }

        // Apply resistance force scaled to spring stiffness
        // const boundaryResistanceFactor = 3.0;
        // const resistanceForce = overpan * this.#currentSpringConfig.stiffness * this.#boundaryResistanceFactor;
        
        // Apply resistance by adjusting target velocity
        const velocityAdjustment = resistanceForce * dt;
        this.#inputVelocity += velocityAdjustment * 0.1; // Small factor to prevent oscillation
      }
    }

      const { min: minLimit, max: maxLimit } = this.domainLimits;
      const currentWidth = this.#physicsCurrent[1] - this.#physicsCurrent[0];
  
      if (this.#physicsCurrent[0] < minLimit) {

        this.#physicsCurrent[1] = minLimit + currentWidth;
      } else if (this.#physicsCurrent[1] > maxLimit) {
        this.#physicsCurrent[1] = maxLimit;
        this.#physicsCurrent[0] = maxLimit - currentWidth;
      }
    }

    #tickSpring(ctx, last_value, current_value, target_value) {
      const delta = target_value - current_value;
      const velocity = (current_value - last_value) / (ctx.dt || 1 / 60);
  
      // if (this.#activePanContext && Math.abs(delta) > 0.001) {
      //   console.log('Pan spring:', {

      //     damping: ctx.opts.damping
      //   });
      // }
  
      const spring = ctx.opts.stiffness * delta;
      const damper = ctx.opts.damping * velocity;
      const acceleration = (spring - damper) * ctx.inv_mass;
      const d = (velocity + acceleration) * ctx.dt;
      
      if (Math.abs(d) < ctx.opts.precision && Math.abs(delta) < ctx.opts.precision) {
        return target_value; // settled
      } else {
        ctx.settled = false; // signal to keep animating
        return current_value + d;
      }
    }

    resetZoom() {

      // Handle invalid extents
      if (targetExtent[0] === targetExtent[1] || 
          !isFinite(targetExtent[0]) || 
          !isFinite(targetExtent[1])) {
        console.warn(`⚠️ Scale ${this.name} skipping zoom - invalid extent:`, targetExtent);
        return;
      }
  
      // This is the final state we want to animate TOWARDS.
      this.#target[0] = targetExtent[0];
      this.#target[1] = targetExtent[1];
      
      if (!this.ready && this.zoomInEffect.enabled) {
        // --- Initial "Zoom-In" Effect ---
        this.#updateSpringConfig(this.zoomInEffect.springConfig);
        
        const center = (targetExtent[0] + targetExtent[1]) / 2;
        const range = targetExtent[1] - targetExtent[0];
        const expandedMin = center - (range * this.zoomInEffect.factor) / 2;
        const expandedMax = center + (range * this.zoomInEffect.factor) / 2;
        
        // Instantly set the starting physics state to the expanded view.
        this.#physicsCurrent[0] = expandedMin;
        this.#physicsCurrent[1] = expandedMax;
        this.#physicsPrevious[0] = expandedMin;
        this.#physicsPrevious[1] = expandedMax;
        this.#renderCurrent[0] = expandedMin;
        this.#renderCurrent[1] = expandedMax;
        
      } else if (!this.ready) {
        // --- Not ready, no animation ---
        // Instantly snap ALL states to the target.
        const [min, max] = targetExtent;

        this.#physicsCurrent[1] = max;
        this.#physicsPrevious[0] = min;
        this.#physicsPrevious[1] = max;
        this.#renderCurrent[0] = min;
        this.#renderCurrent[1] = max;
        
      } else {
        // --- Normal animated reset ---
        // This is called when the user double-clicks. The animation starts
        // from wherever the physics state currently is.
        this.#updateSpringConfig(springConfigs.reset);
      }
  
      // Reset velocity to ensure a smooth start and stop any current motion.
      this.#velocity = [0, 0];
      
      // Ensure the animation loop starts.
      this.#ensureAnimating();
    }
  
    resetDomainLimits() {
      const [min, max] = this.effectiveExtent;
      const width = max - min;
  
      this.domainLimits.min = min - this.zoomLimits.max * width
      this.domainLimits.max = max + this.zoomLimits.max * width
      this.domainLimits.minWidth = width / this.zoomLimits.min;
    }
        
    zoomToRange([min, max]) {
      this.#updateSpringConfig(springConfigs.default);
      this.#target = [min, max];
      this.#ensureAnimating();
    }

        const [min, max] = this.#renderCurrent;
        const range = max - min;
  
        const state = this.#gpuStateObject;
        state.domainMin = min;
        state.domainMax = max;
        // categoryCount is already 0
        state.domainScale = range === 0 ? 0 : 1 / range;
  
        // console.log('gpuState', state.domainMin)
  
        return state;
    }

      this.#frozenExtent = null;
      this.#isAnimating = false;
      this.#resetAccumulator();
      this.#gpuCoordinator.unregisterScale(this.id)
    }

    // #scaleUsage = $state(new Map());
  
    constructor() {
      this.registryId = Math.random().toString(36).slice(2, 8);
      console.log(`🏗️ NEW ScaleRegistry created: ${this.registryId}`);
  
      this.scales = {
        linear: new Map(),
        band: new Map(),
        ordinal: new Map(),
      }
  
      this.scaleUsage = $state(new Map());
    }
  
    getOrCreateScale(name, type='linear', config = {}, chartId = null, gpuCoordinator = null) {
      // Create unique key that includes configuration
      const configKey = Object.keys(config).length > 0 ? `-${JSON.stringify(config)}` : '';
      const key = `${name}-${type}${configKey}`;
      
      if (!this.scales[type].has(key)) {
        console.log(`🆕 Registry ${this.registryId} creating NEW scale: ${name} (${type})`, {
          config: Object.keys(config).length > 0 ? config : 'default'
        });
        
        if (type === 'linear') {
          this.scales[type].set(key, new LinearScaleState(name, undefined, type, undefined, config, gpuCoordinator));
        } else if (type === 'band') {
          this.scales[type].set(key, new BandScaleState(name, [], config));
        } else if (type === 'ordinal') {
          // TODO: fix color scale support, simple default for penguins dataset:
          this.scales[type].set(key, scaleOrdinal()
            .range(['#EFB605', '#FFF84A', '#FF0266'])
            .domain(['Adelie', 'Gentoo', 'Chinstrap']));
        }
      } else {
        console.log(`♻️ Registry ${this.registryId} reusing existing scale: ${name} (${type})`);
      }
      const scale = this.scales[type].get(key);
  
      if (chartId && scale) {
        if (!this.scaleUsage.has(scale)) {
          this.scaleUsage.set(scale, new Set());
        }
        this.scaleUsage.get(scale).add(chartId);
        console.log('scale usage from registration', this.scaleUsage)
      }
  
      return scale;
      // return this.scales[type].get(key);
    }
  
    removeChartUsage(chartId) {
      for (const [scale, chartIds] of this.scaleUsage) {
        chartIds.delete(chartId);
        if (chartIds.size === 0) {
          this.scaleUsage.delete(scale);
        }
      }
    }
  }
  
  
  
  // Create and set the Scale Registry context
  export function setScaleRegistry() {
    const state = new ScaleRegistry();
    return setContext('scale-registry', state);
  }
  
  // Retrieve a specific scale with configuration support
  export function getScaleState(name, type='linear', config = {}, gpuCoordinator = null, chartId = null) {
    const scaleRegistry = getContext('scale-registry');
    
    if (!scaleRegistry) {
      console.error('Scale Registry context missing. Make sure to call setScaleRegistry() in a parent component.');
      return null;
    }
  
    const scaleInstance = scaleRegistry.getOrCreateScale(name, type, config, chartId, gpuCoordinator);
    
    return scaleInstance;
  }

  <script>
    import { onMount, onDestroy, setContext, getContext } from "svelte";
    import { getGPUContext } from '$lib/gpu';
  
    let {
      canvasComponents,
      // msaaSampleCount = 4 
    } = $props();

    let width = $derived(c.innerWidth);
    let height = $derived(c.innerHeight);
    let margin = $derived(c.margin);

    let context;
    let gpuCoordinator;
    let chartControl;
    let contextConfigured = $state(false);
    let initializationError = $state(null);
  
    const chartId = c.chartId;
  
    // Enhanced initialization with comprehensive error handling
    async function initWebGPU() {
      try {      
        // Get shared GPU coordinator
        gpuCoordinator = getGPUContext();
        
        if (!gpuCoordinator) {
          throw new Error('GPUCoordinator not found. Ensure setCoordinators() was called in a parent component.');
        }
  
        // Wait for coordinator initialization with timeout
        let attempts = 0;
        const maxAttempts = 100; // 5 seconds at 50ms intervals
        
        while (!gpuCoordinator.isInitialized && attempts < maxAttempts) {
          await new Promise(resolve => setTimeout(resolve, 50));
          attempts++;
        }
  
        if (!gpuCoordinator.isInitialized) {
          throw new Error(`GPUCoordinator failed to initialize within ${maxAttempts * 50}ms timeout`);
        }
  
        // Configure context with proper error handling
        await configureContext();
        
        // Register with coordinator after context is ready
        registerWithCoordinator();
        
        console.log(`✅ WebGPUCanvas ${chartId} initialized successfully`);
      } catch (error) {
        console.error(`❌ WebGPUCanvas ${chartId} initialization failed:`, error);
        initializationError = error.message;
        throw error;
      }
    }
  
    // Separate context configuration with proper error handling
    async function configureContext() {
      if (!canvas) {
        throw new Error('Canvas element not available for context configuration');
      }
  
      try {
        // Get WebGPU context
        context = canvas.getContext('webgpu', {willReadFrequently: false});
        if (!context) {
          throw new Error('Failed to get WebGPU context from canvas. WebGPU may not be supported.');
        }
  
        const device = gpuCoordinator.device;
        const presentationFormat = gpuCoordinator.presentationFormat;
  
        if (!device) {
          throw new Error('No WebGPU device available from coordinator');
        }
  
        // Configure context
        context.configure({
          device,
          format: presentationFormat,
          alphaMode: 'premultiplied',
          desynchronized: true,
        });
        
        contextConfigured = true;
        
        // Setup initial canvas size
        setupCanvas();
      } catch (error) {
        console.error(`❌ Context configuration failed for chart ${chartId}:`, error);
        contextConfigured = false;
        throw error;
      }
    }
  
    // Clean registration with coordinator
    function registerWithCoordinator() {
      if (!contextConfigured) {
        console.error(`⚠️ Attempting to register chart ${chartId} before context is configured`);
        return;
      }
  
      console.log(`📋 Registering chart ${chartId} with coordinator`);
      
      try {
        chartControl = gpuCoordinator.registerChart(chartId, canvas, {
          zIndex: 0, // zIndex might still be useful for chart-level ordering
        });
  
        // Notify coordinator that context is ready
        chartControl.setContextReady();
        console.log(`✅ Chart ${chartId} registered and ready`);
      } catch (error) {
        console.error(`❌ Chart registration failed for ${chartId}:`, error);
        throw error;
      }
    }
  
    // Robust canvas setup with dimension change detection
    function setupCanvas() {
      if (!canvas || !contextConfigured) {
        console.debug(`⚠️ setupCanvas called before canvas/context ready for ${chartId}`);
        return;
      }
      
      try {
        // Calculate new dimensions
        const devicePixelRatio = window.devicePixelRatio || 1;
  
        const canvasWidth = width * devicePixelRatio;
        const canvasHeight = height * devicePixelRatio;
        
        // Only update if dimensions actually changed to avoid unnecessary work
        const dimensionsChanged = 
          canvas.width !== canvasWidth || 
          canvas.height !== canvasHeight;
        
        const positionChanged = 
          canvas.style.transform !== `translate(${margin.left}px, ${margin.top}px)`;
  
        // Only proceed if there's an actual change.
        if (!dimensionsChanged && !positionChanged) return;
  
        if (dimensionsChanged) {
          canvas.width = canvasWidth;
          canvas.height = canvasHeight;
          canvas.style.width = width + 'px';
          canvas.style.height = height + 'px';        
        }
  
        if (positionChanged) {
          canvas.style.transform = `translate(${margin.left}px, ${margin.top}px)`;
        }
  
        if (chartControl) {
          chartControl.invalidate();
        }
        
      } catch (error) {
        console.error(`❌ Canvas setup failed for ${chartId}:`, error);
        initializationError = `Canvas setup failed: ${error.message}`;
      }
    }

        return { 
          width: width * currentDPR, 
          height: height * currentDPR,
          cssWidth: width,
          cssHeight: height,
          currentDPR

      },
  
      getPresentationFormat() {
        return gpuCoordinator?.presentationFormat;
      },
  
      isInitialized() {
        return contextConfigured && gpuCoordinator?.isInitialized;
      },
      
      invalidate() {
        chartControl?.invalidate();
      },
  
      // Enhanced debugging support
      getDebugInfo() {
        return {
          chartId,
          contextConfigured,
          initializationError,
          layerCount: layers.length,
          layers: layers.map(l => ({
            id: l.id,
            markType: l.markType,
            zIndex: l.zIndex,
            visible: l.visible

            height: canvas?.height,
            cssWidth: canvas?.style.width,
            cssHeight: canvas?.style.height
          }
        };
      }
    });
  
    onMount(async () => {
      try {
        await initWebGPU();
      } catch (error) {
        console.error(`❌ WebGPUCanvas ${chartId} mount failed:`, error);
        // Error is already stored in initializationError state
      }
    });
  
    onDestroy(() => {
      console.log(`🧹 Destroying WebGPUCanvas ${chartId}`);
      
      try {
        // Clean up chart registration
        if (chartControl) {
          chartControl.remove();
          chartControl = null;
        }
        
        // Context cleanup happens automatically when canvas is removed
        contextConfigured = false;
        
        // Clear references
        gpuCoordinator = null;
        context = null;
        
        console.log(`✅ WebGPUCanvas ${chartId} destroyed successfully`);
      } catch (error) {
        console.error(`❌ Error during WebGPUCanvas ${chartId} destruction:`, error);
      }
    });
  
    $effect(() => {
      // Dependencies that trigger a canvas resize/reposition.
      void width;
      void height;
      void margin;
  
      // Only run if the canvas and context are fully ready.
      if (canvas && contextConfigured) {
          console.log(`[${chartId}] Canvas dimension/margin effect triggered.`);
          // The setupCanvas function already contains the logic to check
          // if dimensions *actually* changed before doing expensive work.
          setupCanvas();
          //Tell the coordinator to update this chart's frame uniforms
          if (gpuCoordinator) {
            gpuCoordinator.updateChartFrameUniforms(chartId);
            gpuCoordinator.invalidateChart(chartId);
          }
      }
    });

  // BaseGPURenderer.svelte.js - Simplified for Pipeline Factory Pattern
  // Scale management completely removed - coordinator handles all scale concerns
  
  export class BaseGPURenderer {
    // Core properties
    #markType;
    #webGPUContext;
    #gpuCoordinator;
    #chartContext;
  
    // Resource tracking
    #ownedResources = {
      buffers: [],
      layerControl: null
    };
  
    // State management
    #resourcesCreated = $state(false);
    #initializationError = $state(null);
    #instanceCount = $state(0);
  
    #device = null;
    #pipeline = $state(null); // Provided by coordinator factory
    
    #perDataBindGroup = $state(null);   // @group(1): mark data buffers
    
    #perDataLayout = null;
    #dataBindingConfig = null;
  
    #zIndex;
    #xScaleType;
    #yScaleType;
  
    constructor(markType, webGPUContext, gpuCoordinator, chartContext, options = {}) {
      this.#markType = markType;
      this.#webGPUContext = webGPUContext;
      this.#gpuCoordinator = gpuCoordinator;
      this.#chartContext = chartContext;
      this.#zIndex = options.zIndex || 0;
      
      // Store scale types for coordinator pipeline factory
      this.#xScaleType = options.xScaleType || 'linear';
      this.#yScaleType = options.yScaleType || 'linear';
    }
  
    // ===== PUBLIC GETTERS =====
  
    get resourcesCreated() { return this.#resourcesCreated; }
    get initializationError() { return this.#initializationError; }
    get instanceCount() { return this.#instanceCount; }
    get pipeline() { return this.#pipeline; }
    get markType() { return this.#markType; }  
    get perDataBindGroup() { return this.#perDataBindGroup; }
  
    // ===== INITIALIZATION =====
  
    async initialize(config) {
      try {
        const { dataBindingConfig, markShaderCode, vertexBufferLayout } = config;
        
        if (!dataBindingConfig) {
          throw new Error(`${this.#markType} renderer requires dataBindingConfig`);
        }
  
        console.log(`Initializing ${this.#markType} renderer with pipeline factory pattern`);
        
        this.#device = this.#webGPUContext.getDevice();
        if (!this.#device) {
          throw new Error('No WebGPU device available from context');
        }
  
        this.#dataBindingConfig = dataBindingConfig;
  
        // Get pipeline from coordinator factory (coordinator handles shader generation)
        await this.#getPipelineFromCoordinator(markShaderCode, vertexBufferLayout);
        
        this.#resourcesCreated = true;
        console.log(`${this.#markType} renderer initialized - coordinator provides pipeline and scale management`);
        
      } catch (error) {
        console.error(`${this.#markType} renderer initialization failed:`, error);
        this.#initializationError = error.message;
        throw error;
      }
    }
  
    // ===== PIPELINE FACTORY INTEGRATION =====
  
    async #getPipelineFromCoordinator(markShaderCode, vertexBufferLayout) {
      const scaleTypes = {
        x: this.#chartContext.scaleTypes?.x || this.#xScaleType,
        y: this.#chartContext.scaleTypes?.y || this.#yScaleType
      };
  
      // Use coordinator's pipeline factory
      // Get both pipeline and layouts from coordinator
      const result = await this.#gpuCoordinator.createPipelineForMarkType(
        this.#markType,
        scaleTypes,
        markShaderCode,
        vertexBufferLayout,
        this.#dataBindingConfig // Pass this so coordinator can create stable layouts
      );
  
      console.log('Pipeline factory result:', {
        hasPipeline: !!result.pipeline,
        hasFrameLayout: !!result.frameLayout,
        hasDataLayout: !!result.dataLayout
      });
  
      this.#pipeline = result.pipeline;
      this.#perDataLayout = result.dataLayout;     // From coordinator
  
      if (!this.#pipeline) {
        throw new Error(`Coordinator failed to provide pipeline for ${this.#markType}`);
      }
  
      console.log(`Pipeline received from coordinator for ${this.#markType}`);
    }
  
    createDataBindGroup(dataBindings) {
      if (!this.#device || !this.#perDataLayout) {
        console.warn(`Cannot create per-data bind group for ${this.#markType} - missing device or layout`);
        return;
      }
  
      if (!Array.isArray(dataBindings) || dataBindings.length === 0) {
        console.warn(`Cannot create per-data bind group for ${this.#markType} - invalid dataBindings`);
        return;
      }
  
      try {
        // Validate that all required bindings are provided
        const expectedBindings = this.#dataBindingConfig.map(config => config.binding);
        const providedBindings = dataBindings.map(binding => binding.binding);
        
        for (const expectedBinding of expectedBindings) {
          if (!providedBindings.includes(expectedBinding)) {
            throw new Error(`Missing data binding ${expectedBinding} for ${this.#markType}`);
          }
        }
  
        this.#perDataBindGroup = this.#device.createBindGroup({
          label: `${this.#markType}-per-data-bind-group`,
          layout: this.#perDataLayout,
          entries: dataBindings
        });
  
        console.debug(`Per-data bind group (@group(1)) created for ${this.#markType} with ${dataBindings.length} bindings`);
        
      } catch (error) {
        console.error(`Failed to create per-data bind group for ${this.#markType}:`, error);
        this.#initializationError = error.message;
        throw error;
      }
    }
  
    // ===== BUFFER MANAGEMENT =====
  
    createOrUpdateBuffer(existingBuffer, data, usage = GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST) {
      if (!this.#device) {
        throw new Error('No WebGPU device available');
      }
      
      const requiredSize = data.byteLength;
      
      // Smart buffer resizing to prevent thrashing
      if (!existingBuffer || existingBuffer.size < requiredSize) {
        if (existingBuffer) {
          const index = this.#ownedResources.buffers.indexOf(existingBuffer);
          if (index !== -1) {
            this.#ownedResources.buffers.splice(index, 1);
          }
          existingBuffer.destroy();
        }
  
        // Smart sizing with headroom
        const MEMORY_BLOCK_SIZE = 256;
        const desiredSize = Math.ceil(requiredSize * 1.5);
        const roundedUpSize = Math.ceil(desiredSize / MEMORY_BLOCK_SIZE) * MEMORY_BLOCK_SIZE;
        let bufferSize = Math.max(requiredSize, roundedUpSize);
        bufferSize = (bufferSize + 3) & ~3; // 4-byte alignment
  
        const newBuffer = this.#device.createBuffer({
          size: bufferSize,
          usage,
          label: `${this.#markType}-data-buffer-${this.#ownedResources.buffers.length}`
        });
        
        this.#ownedResources.buffers.push(newBuffer);
        console.debug(`Created ${this.#markType} buffer of size ${bufferSize} bytes (required: ${requiredSize})`);
        return newBuffer;
      }
      
      return existingBuffer;
    }
  
    updateInstanceCount(count) {
      this.#instanceCount = count;
      this.invalidateLayer();
    }
  
    registerLayer(drawFunction) {
      try {
        const chartId = this.#chartContext.chartId;
  
        const layerConfig = {
          draw: drawFunction,
          markType: this.#markType,
          renderer: this, 
          componentId: `${this.#markType}-${crypto.randomUUID().slice(0, 6)}`
        };
  
        this.#ownedResources.layerControl = this.#gpuCoordinator.registerLayer(chartId, layerConfig);
  
        console.log(`${this.#markType} layer registered with coordinator for chart ${chartId}`);
        return this.#ownedResources.layerControl;
  
      } catch (error) {
        console.error(`${this.#markType} layer registration failed:`, error);
        this.#initializationError = error.message;
        throw error;
      }
    }
  
    invalidateLayer() {
      if (this.#gpuCoordinator && this.#chartContext?.chartId) {
        this.#gpuCoordinator.invalidateChart(this.#chartContext.chartId);
      }
    }
  
    // ===== DATA PROCESSING HELPERS =====
  
    processDataForAxis(data, axis) {
      const scaleType = axis === 'x' ? this.#xScaleType : this.#yScaleType;
      const scaleState = axis === 'x' ? this.#chartContext.xScaleState : this.#chartContext.yScaleState;
  
      switch (scaleType) {
        case 'linear':
        case 'identity':
          return new Float32Array(data.map(d => d ?? 0));
          
        case 'band':
        case 'point':
          return scaleState.processDataForGPU(data, this.#markType);
          
        case 'log':
          return new Float32Array(data.map(d => Math.log(Math.max(d || 1, 1))));
          
        case 'time':
          return new Float32Array(data.map(d => d instanceof Date ? d.getTime() : d || 0));
          
        default:
          console.warn(`Unknown scale type: ${scaleType}, treating as linear`);
          return new Float32Array(data.map(d => d ?? 0));
      }
    }
  
    // ===== UTILITY METHODS =====
  
    parseColor(color) {
      try {
        if (typeof color === 'string' && color.startsWith('#')) {
          const r = parseInt(color.slice(1, 3), 16) / 255;
          const g = parseInt(color.slice(3, 5), 16) / 255;
          const b = parseInt(color.slice(5, 7), 16) / 255;
          return [r, g, b, 1.0];
        }
        console.warn(`Unrecognized color format: ${color}, using default blue`);
        return [0.0, 0.0, 1.0, 1.0];
      } catch (error) {
        console.error(`Color parsing failed for ${this.#markType}:`, error);
        return [0.0, 0.0, 1.0, 1.0];
      }
    }
  
    // ===== CLEANUP =====
  
    destroy() {
      console.log(`Cleaning up ${this.#markType} renderer resources`);
      
      try {
        // Clean up owned buffers
        this.#ownedResources.buffers.forEach(buffer => {
          if (buffer && !buffer.destroyed) {
            buffer.destroy();
          }
        });
        this.#ownedResources.buffers = [];
  
        // Clean up layer registration
        if (this.#ownedResources.layerControl) {
          this.#ownedResources.layerControl.remove();
        }
  
        console.log(`${this.#markType} renderer cleanup complete`);
        
      } catch (error) {
        console.error(`Error during ${this.#markType} renderer cleanup:`, error);
      }
    }
  }

  import { untrack } from 'svelte';

  /**
   * A generic controller that handles the common lifecycle, state management,
   * and data fetching orchestration for a mark component using reactive patterns.
   * This class acts as the "ViewModel" or "Controller" for a mark.
   */
  export class BaseMarkSetup {
    // --- Reactive State (for the View to consume) ---
    client = $state(null);
    categoricalDataFetched = $state(false);
  
    // --- Reactive Setup Progress State ---
    #scaleStatesReady = $state(false);
    #scaleMappingsReady = $state(false);
    #clientCreated = $state(false);
    #clientDataReady = $state(false);
    #continuousDomainsSet = $state(false);
    #categoricalFetchStarted = $state(false);
  
    // --- Private Properties ---
    #config;
    #coordinator;
    #chartContext;
    #markId;
    #scaleMap = {};
  
    /**
     * @param {object} config The mark's configuration object, derived from Svelte props.
     * @param {import('@uwdata/mosaic-core').Coordinator} coordinator The Mosaic coordinator instance.
     * @param {object} chartContext The chart's context object.
     */
    constructor(config, coordinator, chartContext, stableId = null) {
      this.#config = config;
      this.#coordinator = coordinator;
      this.#chartContext = chartContext;
      this.#markId =  stableId || `${config.markType}-${crypto.randomUUID().slice(0, 8)}`;
      this.#registerScaleRequirements();
      this.#setupReactiveEffects();
    }
  
    #setupReactiveEffects() {
  
      // Effect 1: Wait for scale objects to be available in context
      $effect(() => {
        const { scaleTypes, xScaleState, yScaleState } = this.#chartContext;
        if (!!(scaleTypes && xScaleState && yScaleState) && !this.#scaleStatesReady) {
          this.#scaleStatesReady = true;
        }
      });
  
      // Effect 2: Setup scale mappings when scale states are ready
      $effect(() => {
        if (this.#scaleStatesReady && !this.#scaleMappingsReady) {
          this.#setupScaleMappings();
          this.#scaleMappingsReady = true;
        }
      });
  
      // Effect 3: Create client when scale mappings are ready
      $effect(() => {
        if (this.#scaleMappingsReady && !this.#clientCreated) {
          const fieldRequests = this.#buildFieldInfoRequests();
          this.#createClient(fieldRequests);
          this.#clientCreated = true;
        }
      });
  
      // Effect 4: Monitor client data readiness
      $effect(() => {
        if (this.#clientCreated && this.client?.fieldInfo && !this.#clientDataReady) {
          this.#clientDataReady = true;
        }
      });
  
      // Effect 5: Set initial continuous domains when client data is ready
      $effect(() => {
        if (this.#clientDataReady && !this.#continuousDomainsSet) {
          this.#setInitialContinuousDomains(this.client.fieldInfo);
          this.#continuousDomainsSet = true;
        }
      });
  
      // Effect 6: Start categorical domain fetch when continuous domains are set
      $effect(() => {
        if (this.#continuousDomainsSet && !this.#categoricalFetchStarted) {
          this.#categoricalFetchStarted = true;
          
          // Trigger async operation without awaiting in effect
          this.#fetchCategoricalDomains().then(() => {
            this.categoricalDataFetched = true;
            this.#setScalesReady();
            console.log(`✅ Mark ${this.#markId} setup complete`);
          }).catch(error => {
            console.error(`❌ Categorical domain fetch failed for ${this.#markId}:`, error);
          });
        }
      });
    }

    /**
     * @virtual
     * Builds the SQL query for the DataClient.
     */
    buildQuery(sourceTable, config, filter) {
      const { select, sort, limit, _filterNulls } = config;
  
      const q = Query.from(sourceTable);
      const selectDef = {};
      const nonAggregateGroupBy = [];
      let hasAggregates = false;
  
      for (const alias in select) {
        const expr = select[alias];
        if (expr == null) continue;
  
        const exprNode = isNode(expr) ? expr : sqlColumn(String(expr));
        selectDef[alias] = exprNode;
  
        if (isAggregateExpression(exprNode)) {
          hasAggregates = true;
        } else {
          nonAggregateGroupBy.push(exprNode);
        }
      }
      q.select(selectDef);
  
      if (hasAggregates && nonAggregateGroupBy.length > 0) {
        q.groupby(nonAggregateGroupBy);
      }
  
      const allFilters = [...filter];
      if (_filterNulls && Array.isArray(_filterNulls)) {
        allFilters.push(..._filterNulls.map(col => isNotNull(col)));
      }
      if (allFilters.length > 0) {
        q.where(and(allFilters));
      }
      
      if (sort) q.orderby({ [sort.by]: sort.direction === 'desc' ? 'DESC' : 'ASC' });
      if (limit) q.limit(limit);
  
      return q;
    }

          queryBuilder: (table, config, filter) => this.buildQuery(table, config, filter),
          dataFormat: 'arrow',
          name: this.#markId,
          ...this.#config.clientOptions
        },
        fieldRequests
      );

      const requests = [];
      const processedFields = new Set();
  
      for (const alias in select) {
        const fieldName = select[alias];
        if (fieldName == null || processedFields.has(fieldName.toString())) continue;
  
        const exprNode = isNode(fieldName) ? fieldName : sqlColumn(String(fieldName));
        if (isAggregateExpression(exprNode)) continue;
  
        const scaleType = alias === 'x' ? this.actualXScaleType : alias === 'y' ? this.actualYScaleType : null;
  
        if (scaleType === 'linear' || scaleType === 'time' || scaleType === 'log') {
          requests.push({ table, column: exprNode, stats: ['min', 'max'] });
          processedFields.add(fieldName.toString());
        }
      }
      return requests;
    }
    
    #setInitialContinuousDomains(fieldInfo) {
      if (!fieldInfo) return;
  
      for (const alias in this.#scaleMap) {
        const scaleState = this.#scaleMap[alias];
        if (!scaleState) continue;
  
        const scaleType = alias === 'x' ? this.actualXScaleType : this.actualYScaleType;
        
        if (scaleType === 'linear' || scaleType === 'time' || scaleType === 'log') {
          const fieldName = this.#config.select[alias];
          if (fieldName == null) continue;
  
          const infoKey = String(fieldName).replace(/^["']|["']$/g, '');
          const info = fieldInfo[infoKey];
  
          if (info && info.min != null && info.max != null) {
            scaleState.addMark([info.min, info.max], this.#markId);
          }
        }
      }
    }
  
    async #fetchCategoricalDomains() {
      const { table, selection, sort } = this.#config;
      if (!this.#coordinator || !table) return;
  
      const fetches = [];
  
      const setupFetch = (axis, field) => {
        if (!field) return;
        const scaleState = this.#chartContext[`${axis}ScaleState`];
        let q = Query.from(table).select({ [axis]: sqlColumn(String(field)) });
  
        // Handle sorting for categorical domains
        const sortConfig = sort;
        if (sortConfig && sortConfig.by) {
          const isSortingByOtherAxis = sortConfig.by === (axis === 'x' ? 'y' : 'x');
          const otherAxisField = this.#config.select[sortConfig.by];
          if (isSortingByOtherAxis && isAggregateExpression(otherAxisField)) {
            q = q.groupby(field).select({ [sortConfig.by]: otherAxisField });
          }
          q.orderby({ [sortConfig.by]: sortConfig.direction === 'desc' ? 'DESC' : 'ASC' });
        } else {
          q.distinct();
        }
  
        fetches.push(
          this.#coordinator.query(q).then(result => {
            const categories = Array.from(result).map(row => row[axis]);
            if (categories.length > 0) {
              scaleState.addMark(categories, this.#markId);
            }
          })
        );
      };
  
      if (this.actualXScaleType === 'band' || this.actualXScaleType === 'point') {
        setupFetch('x', this.#config.select.x);
      }
      if (this.actualYScaleType === 'band' || this.actualYScaleType === 'point') {
        setupFetch('y', this.#config.select.y);
      }
  
      if (fetches.length > 0) await Promise.all(fetches);
    }
  
    #setScalesReady() {
      Object.values(this.#scaleMap).forEach(scale => scale?.setReady?.());
    }
  }
