  /**
   * Rolling average calculator for performance metrics.
   * Filters out negative/invalid values (important for GPU timestamps).
   */
  class NonNegativeRollingAverage {
  	#total = 0;
  	#samples = [];
  	#cursor = 0;
  	#numSamples;
  
  	constructor(numSamples = 120) {
  		this.#numSamples = numSamples;
  	}
  
  	addSample(v) {
  		if (!Number.isNaN(v) && Number.isFinite(v) && v >= 0) {
  			this.#total += v - (this.#samples[this.#cursor] || 0);
  			this.#samples[this.#cursor] = v;
  			this.#cursor = (this.#cursor + 1) % this.#numSamples;
  		}
  	}
  
  	get() {
  		return this.#samples.length > 0 ? this.#total / this.#samples.length : 0;
  	}
  }
  
  /**
   * Manages mapping from scale IDs to SSBO indices.
   * Rebuilds indices when scale topology changes.
   */
  class ScaleIndexManager {
  	#scaleIds = [];
  	#scaleIdToIndex = new Map();
  
  	rebuild(allCurrentScaleIds) {
  		this.#scaleIds = [...allCurrentScaleIds];
  		this.#scaleIdToIndex.clear();
  
  		this.#scaleIds.forEach((scaleId, index) => {
  			this.#scaleIdToIndex.set(scaleId, index);
  		});
  
  		return this.#scaleIds.length;
  	}
  
  	getIndex(scaleId) {
  		return this.#scaleIdToIndex.get(scaleId);
  	}
  
  	getCount() {
  		return this.#scaleIds.length;
  	}
  
  	getAllScaleIds() {
  		return [...this.#scaleIds];
  	}
  }

  /**

   * Key responsibilities:
   * - WebGPU device/context management
   * - Shared pipeline and resource caching with ref-counting

   * - Scale SSBO management for GPU-based domain transforms
   * - Multi-chart render coordination

  	constructor(config = {}) {
  		this.#msaaSampleCount = config.msaaSampleCount ?? 4;
  		this.#scaleIndexManager = new ScaleIndexManager();
  
  		// Create animation loop with callbacks
  		this.#animationLoop = new AnimationLoop({

  		});

  	// =========================================================================
  	// INITIALIZATION
  	// =========================================================================
  
  	async initialize() {
  		if (this.#initialized) return;
  
  		try {
  			await this.#initWebGPU();
  			await this.#initializeScaleBindGroupLayouts();
  
  			this.#initialized = true;
  			this.#handleScaleTopologyChange();
  
  			console.log('GPUCoordinator initialized with Registry Pattern');
  		} catch (error) {
  			console.error('GPUCoordinator initialization failed:', error);
  			throw error;
  		}
  	}
  
  	async #initWebGPU() {
  		if (typeof window === 'undefined' || typeof navigator === 'undefined') {
  			throw new Error('WebGPU initialization attempted server-side');
  		}
  
  		if (!navigator.gpu) {
  			throw new Error('WebGPU not supported');
  		}
  
  		const adapter = await navigator.gpu.requestAdapter({
  			powerPreference: 'high-performance'
  		});
  
  		if (!adapter) {
  			throw new Error('No appropriate GPUAdapter found');
  		}
  
  		this.#canTimestamp = adapter.features.has('timestamp-query');
  
  		this.#device = await adapter.requestDevice({
  			requiredFeatures: this.#canTimestamp ? ['timestamp-query'] : []
  		});
  
  		this.#presentationFormat = navigator.gpu.getPreferredCanvasFormat();
  
  		// Setup timestamp query resources if available
  		if (this.#canTimestamp) {
  			this.#querySet = this.#device.createQuerySet({
  				type: 'timestamp',
  				count: 2
  			});
  
  			this.#resolveBuffer = this.#device.createBuffer({
  				size: this.#querySet.count * 8,
  				usage: GPUBufferUsage.QUERY_RESOLVE | GPUBufferUsage.COPY_SRC
  			});
  
  			this.#resultBuffer = this.#device.createBuffer({
  				size: this.#resolveBuffer.size,
  				usage: GPUBufferUsage.COPY_DST | GPUBufferUsage.MAP_READ
  			});
  		}
  	}
  
  	// =========================================================================
  	// SCALE SSBO SYSTEM
  	// =========================================================================
  
  	async #initializeScaleBindGroupLayouts() {
  		this.#globalScaleLayout = this.#device.createBindGroupLayout({
  			label: 'global-scale-ssbo-layout',
  			entries: [
  				{
  					binding: 0,
  					visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
  					buffer: { type: 'read-only-storage' }
  				}
  			]
  		});
  
  		this.#createGlobalScaleSSBO();
  
  		// Pre-create bind groups for each slot
  		for (let slot = 0; slot < this.#SSBO_SLOT_COUNT; slot++) {
  			const slotOffset = slot * this.#slotByteSize;
  			this.#globalScaleBindGroups[slot] = this.#device.createBindGroup({
  				label: `global-scale-bind-group-slot-${slot}`,
  				layout: this.#globalScaleLayout,
  				entries: [
  					{
  						binding: 0,
  						resource: {
  							buffer: this.#globalScaleSSBO,
  							offset: slotOffset,
  							size: this.#slotByteSize
  						}
  					}
  				]
  			});
  		}
  	}
  
  	#createGlobalScaleSSBO() {
  		const scaleParamSize = 16; // 4 floats * 4 bytes
  		this.#slotByteSize = this.#maxScaleSlots * scaleParamSize;
  		const totalSSBOSize = this.#slotByteSize * this.#SSBO_SLOT_COUNT;
  
  		this.#globalScaleSSBO = this.#device.createBuffer({
  			size: totalSSBOSize,
  			usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
  			label: 'global-scale-ssbo-ring'
  		});
  
  		// CPU buffer for building scale data (reused each frame)
  		this.#cpuScaleBuffer = new Float32Array(this.#slotByteSize / 4);
  	}
  
  	/**
  	 * Fill CPU buffer with current scale GPU state.
  	 * Zero-allocation hot path.
  	 */
  	#fillAllScales(view) {
  		const len = this.#scaleArray.length;
  		for (let i = 0; i < len; i++) {
  			const item = this.#scaleArray[i];
  			const s = item.instance.gpuState;
  
  			const off = item.index << 2; // * 4 via bitshift
  			view[off] = s.domainMin;
  			view[off + 1] = s.domainMax;
  			view[off + 2] = s.domainScale;
  			view[off + 3] = 0; // padding
  		}
  	}
  
  	// =========================================================================
  	// ANIMATION LOOP CALLBACKS
  	// =========================================================================
  
  	/**
  	 * Physics step - called at fixed rate (360Hz).
  	 * Steps all animated scales.
  	 */
  	#physicsStep = () => {
  		const len = this.#animatedScalesArray.length;
  		for (let i = 0; i < len; i++) {
  			this.#animatedScalesArray[i].step();
  		}
  	};
  
  	/**
  	 * Render frame - called once per RAF with extrapolation time.
  	 * Updates render state and triggers GPU render.
  	 */
  
  	#renderFrame = (extrapolationDt) => {
  		// console.log('=== RENDER FRAME ===');
  		// console.log('animatedScalesArray length:', this.#animatedScalesArray.length);
  
  		// const now = performance.now();
  		// if (this.#lastRenderTime > 0 && now - this.#lastRenderTime < 8) {
  		// 	return;
  		// }
  		// this.#lastRenderTime = now;
  
  		// if (this.#framesInFlight >= this.#maxFramesInFlight) {
  		// 	console.log('Skipping frame - too many in flight:', this.#framesInFlight);
  		// 	return;
  		// }
  
  		// 1. Apply any accumulated input (zoom, etc.)
  		for (let i = 0; i < this.#animatedScalesArray.length; i++) {
  			this.#animatedScalesArray[i]._applyInput?.();
  		}
  
  		// 2. Update render state with extrapolation for ALL scales first
  		const len = this.#animatedScalesArray.length;
  		for (let i = 0; i < len; i++) {
  			this.#animatedScalesArray[i].updateRenderState(extrapolationDt);
  		}
  
  		// DEBUG: Log scale values
  		// if (this.#animatedScalesArray.length > 0) {
  		// 	const s = this.#animatedScalesArray[0];
  		// 	console.log('scale:', s.gpuState.domainMin.toFixed(4), s.gpuState.domainMax.toFixed(4));
  		// }
  
  		// 3. Gather charts to render BEFORE removing rested scales
  		const chartArray = this.#chartsToRenderArray;
  		chartArray.length = 0;
  		const currentRenderId = this.#frameId;
  
  		// Add charts affected by animated scales
  		for (let i = 0; i < this.#animatedScalesArray.length; i++) {
  			const scale = this.#animatedScalesArray[i];
  			const affected = scale._affectedChartsArray;
  
  			if (affected) {
  				const affectedLen = affected.length;
  				for (let j = 0; j < affectedLen; j++) {
  					const chartId = affected[j];
  					const chart = this.#registeredCharts.get(chartId);
  
  					if (chart && chart._lastRenderFrame !== currentRenderId) {
  						if (chart.visible && chart.contextConfigured) {
  							chart._lastRenderFrame = currentRenderId;
  							chartArray.push(chart);
  						}
  					}
  				}
  			}
  		}
  
  		// Add dirty charts (from resize, data updates, etc.)
  		const dirtyLen = this.#dirtyChartsArray.length;
  		for (let i = 0; i < dirtyLen; i++) {
  			const chart = this.#dirtyChartsArray[i];
  
  			if (!chart._isDirty) continue; // Removed or already processed
  			chart._isDirty = false;
  
  			if (chart._lastRenderFrame !== currentRenderId) {
  				if (chart.visible && chart.contextConfigured) {
  					chart._lastRenderFrame = currentRenderId;
  					chartArray.push(chart);
  				}
  			}
  		}
  		this.#dirtyChartsArray.length = 0;
  
  		// console.log(
  		// 	'Charts to render:',
  		// 	chartArray.length,
  		// 	chartArray.map((c) => c.chartId)
  		// );
  
  		// // Log affected charts for each scale
  		// for (let i = 0; i < this.#animatedScalesArray.length; i++) {
  		// 	const scale = this.#animatedScalesArray[i];
  		// 	console.log(`Scale ${scale.name} affected charts:`, scale._affectedChartsArray);
  		// }
  
  		// 4. Remove scales that are at rest (backwards for safe removal)
  		for (let i = this.#animatedScalesArray.length - 1; i >= 0; i--) {
  			const scale = this.#animatedScalesArray[i];
  
  			if (scale.isAtRest()) {
  				scale._markAtRest();
  				// Swap and pop removal (O(1))
  				this.#animatedScales.delete(scale);
  				const last = this.#animatedScalesArray.pop();
  				if (i < this.#animatedScalesArray.length) {
  					this.#animatedScalesArray[i] = last;
  				}
  			}
  		}
  
  		// 5. Render if needed
  		if (chartArray.length > 0) {
  			this.#coordinatedRender(chartArray);
  		}
  	};
  
  	/**
  	 * Idle check - return true if animation should stop.
  	 */
  	#idleCheck = () => {
  		return (
  			this.#animatedScales.size === 0 &&
  			this.#dirtyChartsArray.length === 0 &&
  			this.#hoveredCharts.size === 0
  		);
  	};
  
  	// =========================================================================
  	// RENDER PIPELINE
  	// =========================================================================
  
  	#coordinatedRender(chartsNeedingRender) {
  		const slot = this.#frameId % this.#SSBO_SLOT_COUNT;
  
  		// PHASE 1: EARLY TEXTURE ACQUISITION
  		this.#acquiredTextures.length = 0;
  
  		const numCharts = chartsNeedingRender.length;
  		for (let i = 0; i < numCharts; i++) {
  			const chartConfig = chartsNeedingRender[i];
  			const texture = chartConfig.context.getCurrentTexture();
  			const view = texture.createView();
  			this.#acquiredTextures.push({ chartConfig, texture, view });
  		}
  
  		// PHASE 2: CPU WORK (Scale buffer updates)
  		this.#fillAllScales(this.#cpuScaleBuffer);
  
  		const dstOffset = slot * this.#slotByteSize;
  		this.#device.queue.writeBuffer(
  			this.#globalScaleSSBO,
  			dstOffset,
  			this.#cpuScaleBuffer,
  			0,
  			this.#cpuScaleBuffer.length
  		);
  
  		// PHASE 3: GPU COMMAND ENCODING
  		const commandEncoder = this.#device.createCommandEncoder(this.#encoderDescriptor);
  		const currentScaleBindGroup = this.#globalScaleBindGroups[slot];
  
  		for (let i = 0; i < this.#acquiredTextures.length; i++) {
  			// for (let i = 0; i < numCharts; i++) {
  			const { chartConfig, texture, view } = this.#acquiredTextures[i];
  			// const chartConfig = chartsNeedingRender[i];
  			// const texture = chartConfig.context.getCurrentTexture();
  			// const view = texture.createView();
  
  			if (chartConfig.needsPlanRebuild) {
  				this.#buildRenderPlanForChart(chartConfig);
  			}
  
  			const passEncoder = commandEncoder.beginRenderPass(
  				this.#createRenderPassDescriptor(view, chartConfig)
  				// this.#createRenderPassDescriptor(view, chartConfig.canvas, chartConfig.chartId)
  			);
  
  			if (chartConfig.perFrameBindGroup) {
  				passEncoder.setBindGroup(0, chartConfig.perFrameBindGroup);
  				passEncoder.setBindGroup(2, currentScaleBindGroup);
  
  				const plan = chartConfig.renderPlan;
  				const planLen = plan.length;
  
  				let currentPipeline = null;
  				let currentPerDataBindGroup = null;
  
  				for (let j = 0; j < planLen; j++) {
  					const layer = plan[j];
  					const renderer = layer.renderer;
  
  					if (!renderer || !renderer.perDataBindGroup) continue;
  
  					if (layer.cachedPipeline !== currentPipeline) {
  						passEncoder.setPipeline(layer.cachedPipeline);
  						currentPipeline = layer.cachedPipeline;
  					}
  
  					if (renderer.perDataBindGroup !== currentPerDataBindGroup) {
  						passEncoder.setBindGroup(1, renderer.perDataBindGroup);
  						currentPerDataBindGroup = renderer.perDataBindGroup;
  					}
  
  					layer.draw(passEncoder);
  				}
  			}
  
  			passEncoder.end();
  		}
  
  		// PHASE 4: SUBMIT
  		this.#submitArray[0] = commandEncoder.finish();
  		this.#device.queue.submit(this.#submitArray);
  
  		// this.#framesInFlight++;
  		// this.#device.queue.onSubmittedWorkDone().then(() => {
  		// 	this.#framesInFlight--;
  		// });
  
  		this.#frameId++;
  	}
  
  	// =========================================================================
  	// RESOURCE ACQUISITION (Registry Pattern)
  	// =========================================================================
  
  	/**
  	 * Acquire pipeline resources for a renderer.
  	 * Tracks acquisition in registry for proper cleanup.
  	 *
  	 * @param {string} rendererId - Unique renderer identifier
  	 * @param {string} markType - Mark type (dots, rects, etc.)
  	 * @param {object} scaleTypes - {x: string, y: string}
  	 * @param {string} markShaderCode - Mark-specific shader code
  	 * @param {array} vertexBufferLayout - Vertex buffer layout
  	 * @param {array} dataBindingConfig - Data binding configuration
  	 * @returns {Promise<{pipeline, frameLayout, dataLayout}>}
  	 */
  	async acquirePipeline(
  		rendererId,
  		markType,
  		scaleTypes,
  		markShaderCode,
  		vertexBufferLayout,
  		dataBindingConfig
  	) {
  		if (!rendererId) throw new Error('acquirePipeline requires a rendererId');
  
  		// Generate stable resource key (includes scale topology)
  		const topologySuffix = `${scaleTypes.x}-${scaleTypes.y}`;
  		const resourceKey = `${markType}-${topologySuffix}`;
  
  		// Prepare local registry for this renderer
  		if (!this.#rendererRegistry.has(rendererId)) {
  			this.#rendererRegistry.set(rendererId, new Map());
  		}
  		const localRegistry = this.#rendererRegistry.get(rendererId);
  
  		// --- Acquire Shader (with idempotency check) ---
  		let shaderModule;
  		const shaderLookupKey = `shader:${resourceKey}`;
  
  		if (localRegistry.has(shaderLookupKey)) {
  			// Already owned by this renderer - get reference without incrementing refCount
  			const cached = this.#sharedResources.shaderModules.get(resourceKey);
  			if (!cached) {
  				// Defensive: global cache inconsistent, clear local entry
  				localRegistry.delete(shaderLookupKey);
  			} else {
  				shaderModule = cached.module;
  			}
  		}
  
  		if (!shaderModule) {
  			// Create/Get and increment RefCount
  			const templateResult = scaleTemplateSystem.generateShader(scaleTypes, markShaderCode);
  			shaderModule = await this.#getOrCreateSharedShader(resourceKey, templateResult.shaderCode);
  			localRegistry.set(shaderLookupKey, { type: 'shader', key: resourceKey });
  		}
  
  		// --- Acquire Layouts ---
  		const frameLayout = this.#getOrCreateFrameLayout();
  		const dataLayout = this.#getOrCreateDataLayout(markType, dataBindingConfig);
  
  		// --- Acquire Pipeline (with idempotency check) ---
  		let pipeline;
  		const pipelineLookupKey = `pipeline:${resourceKey}`;
  
  		if (localRegistry.has(pipelineLookupKey)) {
  			const cached = this.#sharedResources.pipelines.get(resourceKey);
  			if (!cached) {
  				localRegistry.delete(pipelineLookupKey);
  			} else {
  				pipeline = cached.pipeline;
  			}
  		}
  
  		if (!pipeline) {
  			// Create/Get and increment RefCount
  			// Note: pipelineLayout created inside callback to avoid waste when cached
  			pipeline = await this.#getOrCreateSharedPipeline(resourceKey, async () => {
  				const pipelineLayout = this.#device.createPipelineLayout({
  					label: `${resourceKey}-pipeline-layout`,
  					bindGroupLayouts: [frameLayout, dataLayout, this.#globalScaleLayout]
  				});
  
  				return this.#device.createRenderPipeline({
  					label: `${resourceKey}-pipeline`,
  					layout: pipelineLayout,
  					vertex: {
  						module: shaderModule,
  						entryPoint: 'vertexMain',
  						buffers: vertexBufferLayout || []
  					},
  					fragment: {
  						module: shaderModule,
  						entryPoint: 'fragmentMain',
  						targets: [
  							{
  								format: this.#presentationFormat,
  								blend: {
  									color: {
  										srcFactor: 'src-alpha',
  										dstFactor: 'one-minus-src-alpha',
  										operation: 'add'
  									},
  									alpha: {
  										srcFactor: 'one',
  										dstFactor: 'one-minus-src-alpha',
  										operation: 'add'
  									}
  								}
  							}
  						]
  					},
  					primitive: { topology: 'triangle-list' },
  					multisample: { count: this.#msaaSampleCount > 1 ? this.#msaaSampleCount : 1 }
  				});
  			});
  
  			localRegistry.set(pipelineLookupKey, { type: 'pipeline', key: resourceKey });
  		}
  
  		return { pipeline, frameLayout, dataLayout };
  	}
  
  	/**
  	 * Release all resources tracked for a specific renderer ID.
  	 * Called by BaseGPURenderer.destroy()
  	 */
  	releaseRenderer(rendererId) {
  		const localRegistry = this.#rendererRegistry.get(rendererId);
  		if (!localRegistry) return;
  
  		for (const res of localRegistry.values()) {
  			if (res.type === 'pipeline') {
  				this.#releaseSharedPipeline(res.key);
  			} else if (res.type === 'shader') {
  				this.#releaseSharedShader(res.key);
  			}
  		}
  
  		this.#rendererRegistry.delete(rendererId);
  	}

  	// INTERNAL REF-COUNTING (Private)
  	// =========================================================================
  
  	async #getOrCreateSharedPipeline(key, createFn) {
  		const existing = this.#sharedResources.pipelines.get(key);
  		if (existing) {
  			existing.refCount++;
  			return existing.pipeline;
  		}
  
  		// Async Lock to prevent double-creation
  		if (this.#creatingPipelines.has(key)) {
  			while (this.#creatingPipelines.has(key)) {
  				await new Promise((r) => setTimeout(r, 10));
  			}
  			// Check again after wait
  			const newExisting = this.#sharedResources.pipelines.get(key);
  			if (newExisting) {
  				newExisting.refCount++;
  				return newExisting.pipeline;
  			}
  		}
  
  		this.#creatingPipelines.add(key);
  		try {
  			const pipeline = await createFn();
  			this.#sharedResources.pipelines.set(key, {
  				pipeline,
  				refCount: 1,
  				createdAt: Date.now()
  			});
  			return pipeline;
  		} finally {
  			this.#creatingPipelines.delete(key);
  		}
  	}
  
  	#releaseSharedPipeline(key) {
  		const resource = this.#sharedResources.pipelines.get(key);
  		if (!resource) return;
  
  		resource.refCount--;
  		if (resource.refCount <= 0) {
  			this.#sharedResources.pipelines.delete(key);
  		}
  	}
  
  	async #getOrCreateSharedShader(key, code) {
  		const existing = this.#sharedResources.shaderModules.get(key);
  		if (existing) {
  			existing.refCount++;
  			return existing.module;
  		}
  
  		const module = this.#device.createShaderModule({ code });
  		this.#sharedResources.shaderModules.set(key, {
  			module,
  			refCount: 1,
  			createdAt: Date.now()
  		});
  		return module;
  	}
  
  	#releaseSharedShader(key) {
  		const resource = this.#sharedResources.shaderModules.get(key);
  		if (!resource) return;
  
  		resource.refCount--;
  		if (resource.refCount <= 0) {
  			this.#sharedResources.shaderModules.delete(key);
  		}
  	}

  	scheduleTopologyUpdate() {
  		if (this.#topologyUpdateScheduled) return;
  		this.#topologyUpdateScheduled = true;
  
  		queueMicrotask(() => {
  			this.#topologyUpdateScheduled = false;
  			this.#handleScaleTopologyChange();
  		});
  	}
  
  	#handleScaleTopologyChange() {
  		if (!this.#scaleRegistry) return;
  
  		const usageMap = this.#scaleRegistry.scaleUsage;
  		if (!usageMap || typeof usageMap[Symbol.iterator] !== 'function') return;
  
  		// Clear and rebuild
  		const ids = [];
  		this.#scaleIdToInstance.clear();
  		this.#scaleArray = [];
  
  		for (const [scaleInstance, chartIds] of usageMap) {
  			if (chartIds && typeof chartIds[Symbol.iterator] === 'function') {
  				scaleInstance._affectedChartsArray = Array.from(chartIds);
  			} else {
  				scaleInstance._affectedChartsArray = [];
  			}
  
  			if (scaleInstance?.id) {
  				ids.push(scaleInstance.id);
  				this.#scaleIdToInstance.set(scaleInstance.id, scaleInstance);
  			}
  		}
  
  		const count = this.#scaleIndexManager.rebuild(ids);
  
  		// Build flat array for hot loop
  		for (const scale of this.#scaleIdToInstance.values()) {
  			const idx = this.#scaleIndexManager.getIndex(scale.id);
  			if (idx !== undefined) {
  				this.#scaleArray.push({ instance: scale, index: idx });
  			}
  		}
  
  		// Grow SSBO if needed
  		if (count > this.#maxScaleSlots) {
  			let newSlots = this.#maxScaleSlots;
  			while (newSlots < count) newSlots *= 2;
  			this.#maxScaleSlots = newSlots;
  			this.#recreateGlobalScalesSSBOAndBindGroups();
  		}
  
  		// Update per-chart uniforms (scale indices may have changed)
  		for (const [chartId] of this.#registeredCharts) {
  			this.updateChartFrameUniforms(chartId);
  		}
  	}
  
  	#recreateGlobalScalesSSBOAndBindGroups() {
  		console.log('Resizing Global Scale SSBO...');
  
  		const oldSSBO = this.#globalScaleSSBO;
  
  		const scaleParamSize = 16;
  		this.#slotByteSize = this.#maxScaleSlots * scaleParamSize;
  		const totalSSBOSize = this.#slotByteSize * this.#SSBO_SLOT_COUNT;
  
  		try {
  			const newSSBO = this.#device.createBuffer({
  				size: totalSSBOSize,
  				usage: GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST,
  				label: `global-scale-ssbo-ring-v${this.#frameId}`
  			});
  
  			const newBindGroups = [];
  			for (let slot = 0; slot < this.#SSBO_SLOT_COUNT; slot++) {
  				const offset = slot * this.#slotByteSize;
  				newBindGroups[slot] = this.#device.createBindGroup({
  					label: `global-scale-bind-group-slot-${slot}`,
  					layout: this.#globalScaleLayout,
  					entries: [
  						{
  							binding: 0,
  							resource: {
  								buffer: newSSBO,
  								offset,
  								size: this.#slotByteSize
  							}
  						}
  					]
  				});
  			}
  
  			this.#globalScaleSSBO = newSSBO;
  			this.#globalScaleBindGroups = newBindGroups;
  			this.#cpuScaleBuffer = new Float32Array(this.#slotByteSize / 4);
  
  			if (oldSSBO) {
  				oldSSBO.destroy();
  			}
  		} catch (error) {
  			console.error('Failed to resize Scale SSBO. Keeping old buffer.', error);
  		}
  	}

  	addAnimatingScale(scale) {
  		// console.log(`=== addAnimatingScale ${scale.name} (${scale.id}) ===`);
  		let idx = this.#scaleIndexManager.getIndex(scale.id);
  		// console.log('Initial index lookup:', idx);
  
  		if (idx === undefined) {
  			// console.log('Index undefined, calling registerScale...');
  			this.registerScale(scale.id, 'linear');
  			idx = this.#scaleIndexManager.getIndex(scale.id);
  			// console.log('Index after registerScale:', idx);
  			if (idx === undefined) {
  				console.error(`Scale ${scale.id} still not registered after auto-register`);
  				return;
  			}
  		}
  
  		// console.log('Already in animatedScales Set?', this.#animatedScales.has(scale));
  
  		if (!this.#animatedScales.has(scale)) {
  			this.#animatedScales.add(scale);
  			this.#animatedScalesArray.push(scale);
  			// console.log('Added to animatedScales, new length:', this.#animatedScalesArray.length);
  		}
  
  		this.#animationLoop.requestStart();
  	}
  
  	removeAnimatingScale(scale) {
  		if (this.#animatedScales.delete(scale)) {
  			const idx = this.#animatedScalesArray.indexOf(scale);
  			if (idx !== -1) {
  				const lastItem = this.#animatedScalesArray.pop();
  				if (idx < this.#animatedScalesArray.length) {
  					this.#animatedScalesArray[idx] = lastItem;
  				}
  			}
  		}
  	}

  	registerChart(chartId, canvas, config) {
  		if (this.#registeredCharts.has(chartId)) {
  			console.warn(`Chart ${chartId} already registered, updating config`);
  		}
  
  		const chartConfig = {
  			chartId,
  			canvas,
  			zIndex: config.zIndex || 0,
  			visible: true,
  			context: canvas.getContext('webgpu'),
  			contextConfigured: false,
  			layers: new Map(),
  			renderPlan: [],
  			needsPlanRebuild: true,
  			xScaleId: null,
  			yScaleId: null,
  			frameUniformsBuffer: null,
  			perFrameBindGroup: null,
  			renderPassDescriptor: null,
  			msaa: null,
  			_lastRenderFrame: -1,
  			_isDirty: false
  		};
  
  		this.#registeredCharts.set(chartId, chartConfig);
  
  		const pendingMapping = this.#pendingScaleMappings.get(chartId);
  		if (pendingMapping) {
  			this.setChartScaleMapping(chartId, pendingMapping);
  			this.#pendingScaleMappings.delete(chartId);
  		}
  
  		return {
  			setContextReady: () => {
  				const chart = this.#registeredCharts.get(chartId);
  				if (chart) {
  					chart.contextConfigured = true;
  					this.#processPendingFrameUpdates(chartId);
  				}
  			},
  			setVisible: (visible) => {
  				const cfg = this.#registeredCharts.get(chartId);
  				if (cfg && cfg.visible !== visible) {
  					cfg.visible = visible;
  					this.invalidateChart(chartId);
  				}
  			},
  			invalidate: () => this.invalidateChart(chartId),
  			remove: () => this.unregisterChart(chartId)
  		};
  	}
  
  	unregisterChart(chartId) {
  		const chartConfig = this.#registeredCharts.get(chartId);
  		if (chartConfig) {
  			chartConfig._isDirty = false;
  
  			if (chartConfig.frameUniformsBuffer) {
  				chartConfig.frameUniformsBuffer.destroy();
  			}
  
  			if (chartConfig.msaa) {
  				chartConfig.msaa.texture.destroy();
  				chartConfig.msaa = null;
  			}
  
  			this.#registeredCharts.delete(chartId);
  			this.#hoveredCharts.delete(chartId);
  
  			// const msaa = this.#msaaCache.get(chartId);
  			// if (msaa) {
  			// 	msaa.texture.destroy();
  			// 	this.#msaaCache.delete(chartId);
  			// }
  
  			this.#pendingScaleMappings.delete(chartId);
  		}
  	}
  
  	invalidateChart(chartId) {
  		const chart = this.#registeredCharts.get(chartId);
  		if (chart && !chart._isDirty) {
  			chart._isDirty = true;
  			this.#dirtyChartsArray.push(chart);
  			this.#animationLoop.requestStart();
  		}
  	}
  
  	setChartScaleMapping(chartId, { xScaleId, yScaleId }) {
  		const chartConfig = this.#registeredCharts.get(chartId);
  		if (!chartConfig) {
  			this.#pendingScaleMappings.set(chartId, { xScaleId, yScaleId });
  			return;
  		}
  
  		const changed = chartConfig.xScaleId !== xScaleId || chartConfig.yScaleId !== yScaleId;
  
  		chartConfig.xScaleId = xScaleId;
  		chartConfig.yScaleId = yScaleId;
  
  		if (changed) {
  			chartConfig.needsPlanRebuild = true;
  			this.invalidateChart(chartId);
  		}
  	}

  	#getOrCreateDataLayout(markType, dataBindingConfig) {
  		const layoutKey = `${markType}-${JSON.stringify(dataBindingConfig)}`;
  
  		if (!this.#dataLayouts.has(layoutKey)) {
  			const layout = this.#device.createBindGroupLayout({
  				label: `${markType}-data-layout`,
  				entries: dataBindingConfig.map((config) => ({
  					binding: config.binding,
  					visibility: GPUShaderStage.VERTEX | GPUShaderStage.FRAGMENT,
  					buffer: { type: config.type || 'read-only-storage' }

  	#createChartFrameResources(chartId) {
  		const chartConfig = this.#registeredCharts.get(chartId);
  		if (!chartConfig || chartConfig.frameUniformsBuffer) return;
  
  		chartConfig.frameUniformsBuffer = this.#device.createBuffer({
  			size: 48,
  			usage: GPUBufferUsage.UNIFORM | GPUBufferUsage.COPY_DST,
  			label: `frame-uniforms-${chartId}`
  		});
  
  		this.#createChartFrameBindGroup(chartId);
  	}
  
  	#createChartFrameBindGroup(chartId) {
  		const chartConfig = this.#registeredCharts.get(chartId);
  		if (!chartConfig?.frameUniformsBuffer || chartConfig.perFrameBindGroup) return;
  
  		const frameLayout = this.#getOrCreateFrameLayout();
  
  		chartConfig.perFrameBindGroup = this.#device.createBindGroup({
  			label: `per-frame-bind-group-${chartId}`,
  			layout: frameLayout,
  			entries: [
  				{
  					binding: 0,
  					resource: { buffer: chartConfig.frameUniformsBuffer }
  				}
  			]
  		});
  	}
  
  	updateChartFrameUniforms(chartId) {
  		const chartConfig = this.#registeredCharts.get(chartId);
  		if (!chartConfig?.contextConfigured) {
  			this.#pendingFrameUpdates.set(chartId, { timestamp: Date.now() });
  			return;
  		}
  
  		if (!chartConfig.frameUniformsBuffer) {
  			this.#createChartFrameResources(chartId);
  		}
  
  		const dpr = window.devicePixelRatio || 1;
  		const canvas = chartConfig.canvas;
  
  		const resX = canvas.width;
  		const resY = canvas.height;
  		const rangeMinX = 0;
  		const rangeMinY = canvas.clientHeight * dpr;
  		const rangeMaxX = canvas.clientWidth * dpr;
  		const rangeMaxY = 0;
  
  		const xScaleIndex = this.#scaleIndexManager.getIndex(chartConfig.xScaleId) ?? 0;
  		const yScaleIndex = this.#scaleIndexManager.getIndex(chartConfig.yScaleId) ?? 1;
  
  		const buf = new ArrayBuffer(48);
  		const f32 = new Float32Array(buf);
  		const u32 = new Uint32Array(buf);
  
  		f32[0] = resX;
  		f32[1] = resY;
  		f32[2] = rangeMinX;
  		f32[3] = rangeMinY;
  		f32[4] = rangeMaxX;
  		f32[5] = rangeMaxY;
  		f32[6] = dpr;
  		f32[7] = 0.0;
  
  		u32[8] = xScaleIndex >>> 0;
  		u32[9] = yScaleIndex >>> 0;
  		u32[10] = 0;
  		u32[11] = 0;
  
  		this.#device.queue.writeBuffer(chartConfig.frameUniformsBuffer, 0, buf);
  	}
  
  	#processPendingFrameUpdates(chartId) {
  		if (this.#pendingFrameUpdates.has(chartId)) {
  			this.updateChartFrameUniforms(chartId);
  			this.#pendingFrameUpdates.delete(chartId);
  		}
  	}

  	registerLayer(chartId, layerConfig) {
  		const chart = this.#registeredCharts.get(chartId);
  		if (!chart) {
  			console.error(`Attempted to register layer on non-existent chart: ${chartId}`);
  			return { remove: () => {} };
  		}
  
  		const layerId = layerConfig.id || `layer-${crypto.randomUUID().slice(0, 8)}`;
  		chart.layers.set(layerId, { ...layerConfig, id: layerId, zIndex: layerConfig.zIndex ?? 0 });
  		chart.needsPlanRebuild = true;
  		this.invalidateChart(chartId);
  
  		return {
  			remove: () => this.unregisterLayer(chartId, layerId)
  		};
  	}
  
  	unregisterLayer(chartId, layerId) {
  		const chart = this.#registeredCharts.get(chartId);
  		if (chart?.layers.has(layerId)) {
  			chart.layers.delete(layerId);
  			chart.needsPlanRebuild = true;
  			this.invalidateChart(chartId);
  		}
  	}
  
  	#buildRenderPlanForChart(chartConfig) {
  		const layers = Array.from(chartConfig.layers.values());
  		if (layers.length === 0) {
  			chartConfig.renderPlan = [];
  			chartConfig.needsPlanRebuild = false;
  			return;
  		}
  
  		const zIndexGroups = new Map();
  		for (const layer of layers) {
  			const zIndex = layer.zIndex;
  			if (!zIndexGroups.has(zIndex)) zIndexGroups.set(zIndex, []);
  			zIndexGroups.get(zIndex).push(layer);
  		}
  
  		const sortedZLevels = Array.from(zIndexGroups.keys()).sort((a, b) => a - b);
  		const newPlan = [];
  
  		for (const zLevel of sortedZLevels) {
  			const layersInZLevel = zIndexGroups.get(zLevel);
  			const markTypeGroups = new Map();
  
  			for (const layer of layersInZLevel) {
  				const markType = layer.markType || 'unknown';
  				if (!markTypeGroups.has(markType)) markTypeGroups.set(markType, []);
  				markTypeGroups.get(markType).push(layer);
  			}
  
  			for (const batchedLayers of markTypeGroups.values()) {
  				newPlan.push(...batchedLayers);
  			}
  		}
  
  		for (const layer of newPlan) {
  			layer.cachedPipeline = layer.renderer?.pipeline;
  		}
  
  		chartConfig.renderPlan = newPlan;
  		chartConfig.needsPlanRebuild = false;
  	}

  		if (!this.#enableUpdates || this.#extents.size === 0) return;
  
  		const extents = Array.from(this.#extents.values());
  		const minStart = Math.min(...extents.map((r) => r[0]));
  		const maxEnd = Math.max(...extents.map((r) => r[1]));
  
  		if (nice) {
  			const niceDomain = scaleLinear().domain([minStart, maxEnd]).nice(5).domain();
  			this.#extent[0] = niceDomain[0];
  			this.#extent[1] = niceDomain[1];
  		} else {
  			this.#extent[0] = minStart;
  			this.#extent[1] = maxEnd;
  		}
  
  		this.resetDomainLimits();
  		this.resetZoom();
  	}

  			? [this.#extent[0], this.#extent[1]]
  			: this.#frozenExtent || [this.#extent[0], this.#extent[1]];

  		const [min, max] = this.effectiveExtent;
  		const w = max - min;
  		this.domainLimits.min = min - this.zoomLimits.max * w;
  		this.domainLimits.max = max + this.zoomLimits.max * w;
  		this.domainLimits.minWidth = w / this.zoomLimits.min;

  		const min = this.#renderMin;
  		const max = this.#renderMax;
  		const range = max - min;
  
  		this.gpuState.domainMin = min;
  		this.gpuState.domainMax = max;
  		this.gpuState.domainScale = range === 0 ? 0 : 1 / range;
  	}

  		if (!this.#isAnimating && this.#gpuCoordinator) {
  			this.#isAnimating = true;
  			this.#gpuCoordinator.addAnimatingScale(this);
  		}
  	}
  
  	#applyZoomToTarget(center, scaleFactor) {
  		const min = this.#springPair.targetA;
  		const max = this.#springPair.targetB;
  		const curWidth = max - min;
  		const newWidth = curWidth / scaleFactor;
  
  		// Clamp to minimum width
  		const safeWidth = Math.max(newWidth, this.domainLimits.minWidth);
  
  		// Maintain zoom center position
  		const ratio = (center - min) / curWidth;
  		let newMin = center - ratio * safeWidth;
  		let newMax = newMin + safeWidth;
  
  		// Clamp to domain limits
  		const clamped = this.#clampDomain(newMin, newMax);
  		this.#springPair.setTarget(clamped.min, clamped.max);
  	}
  
  	#clampDomain(min, max) {
  		const limits = this.domainLimits;
  		const width = max - min;
  		const limitWidth = limits.max - limits.min;
  
  		if (width > limitWidth) {
  			this.#clampResult.min = limits.min;
  			this.#clampResult.max = limits.max;
  			return this.#clampResult;
  		}
  
  		if (min < limits.min) {
  			min = limits.min;
  			max = min + width;
  		}
  		if (max > limits.max) {
  			max = limits.max;
  			min = max - width;
  			if (min < limits.min) {
  				min = limits.min;
  			}
  		}
  
  		this.#clampResult.min = min;
  		this.#clampResult.max = max;
  		return this.#clampResult;
  	}
  
  	// ===========================================================================
  	// CLEANUP
  	// ===========================================================================
  
  	destroy() {
  		this.#isAnimating = false;
  		if (this.#gpuCoordinator?.removeAnimatingScale) {
  			this.#gpuCoordinator.removeAnimatingScale(this);
  		}
  	}
  }

      #physicsStepsThisFrame = 0;
      #lastDelta = 0;
  
      #boundFrame;
  
      constructor({ onPhysicsStep, onRender, onIdleCheck = null }) {
          if (typeof onPhysicsStep !== 'function') {
              throw new Error('AnimationLoop requires onPhysicsStep callback');
          }
          if (typeof onRender !== 'function') {
              throw new Error('AnimationLoop requires onRender callback');
          }
  
          this.#onPhysicsStep = onPhysicsStep;
          this.#onRender = onRender;
          this.#onIdleCheck = onIdleCheck;
  
          this.#boundFrame = this.#frame.bind(this);
      }
  
      // ===========================================================================
      // PUBLIC API
      // ===========================================================================
  
      start() {
          if (this.#rafId !== null) return;
  
          this.#isRunning = true;
          this.#lastTime = performance.now();
          this.#accumulator = 0;
          this.#rafId = requestAnimationFrame(this.#boundFrame);
      }
  
      requestStart() {
          if (this.#rafId === null) {
              this.start();
          }
      }
  
      stop() {
          this.#isRunning = false;
          if (this.#rafId !== null) {
              cancelAnimationFrame(this.#rafId);
              this.#rafId = null;
          }
      }
  
      destroy() {
          this.stop();
      }
  
      // ===========================================================================
      // GETTERS
      // ===========================================================================
  
      get frameCount() {
          return this.#frameCount;
      }
  
      get physicsStepsLastFrame() {
          return this.#physicsStepsThisFrame;
      }
  
      get lastDelta() {
          return this.#lastDelta;
      }
  
      get accumulator() {
          return this.#accumulator;
      }
  
      get isRunning() {
          return this.#isRunning && this.#rafId !== null;
      }
  
      // ===========================================================================
      // FRAME LOOP
      // ===========================================================================
  
      #frame(time) {
        const now = time;
          // const now = performance.now();
          let delta = (now - this.#lastTime) / 1000;
          this.#lastTime = now;
  
          // Clamp delta - handles tab switch, debugger pause, slow frames
          if (delta > MAX_DELTA) {
              delta = MAX_DELTA;
          }
  
          this.#lastDelta = delta;
          this.#accumulator += delta;
  
          // Fixed timestep physics
          this.#physicsStepsThisFrame = 0;
          while (this.#accumulator >= DT) {
              this.#onPhysicsStep();
              this.#accumulator -= DT;
              this.#physicsStepsThisFrame++;
          }
  
          // Render with extrapolation
          this.#onRender(this.#accumulator);
  
          this.#frameCount++;
  
          // Check if we should continue
          const shouldContinue = this.#onIdleCheck ? !this.#onIdleCheck() : true;
  
          if (this.#isRunning && shouldContinue) {
              this.#rafId = requestAnimationFrame(this.#boundFrame);
          } else {
              this.#rafId = null;
          }
      }
  }

  // Fixed timestep for deterministic physics simulation.

  export const PHYSICS_HZ = 360;
  export const DT = 1 / PHYSICS_HZ;           // Fixed timestep in seconds (~2.78ms)
  export const DT_MS = DT * 1000;             // Fixed timestep in milliseconds

  	pan: { stiffness: 600, precision: 1e-6 },

  	reset: { stiffness: 150, precision: 1e-4 },

  // -----------------------------------------------------------------------------
  // SPRING CLASS
  // -----------------------------------------------------------------------------
  /**
   * A critically-damped spring optimized for UI animations.
   *
   * Key features:
   * - Fixed timestep physics for deterministic behavior
   * - Precomputed exponential decay for hot-path performance
   * - Pure render extrapolation (no state mutation)
   * - Zero-allocation in animation loop
   *
   * Usage:
   * ```js
   * const spring = new Spring(0, SPRING_PRESETS.default.stiffness);
   *
   * // In physics loop (fixed timestep):
   * spring.step();
   *
   * // In render loop (variable timestep):
   * const renderValue = spring.getRenderable(remainingDt);
   * ```
   */
  export class Spring {
  	// Current physics state
  	current = 0;
  	target = 0;
  	velocity = 0;
  
  	// Precomputed constants (set via configure())
  	#omega = 0;      // Natural frequency: sqrt(stiffness)
  	#expDt = 0;      // Precomputed: exp(-omega * DT)
  	#precision = 1e-6;
  
  	/**
  	 * @param {number} initial - Initial value
  	 * @param {number} stiffness - Spring stiffness (default: 400)
  	 * @param {number} precision - Rest detection threshold (default: 1e-6)
  	 */
  	constructor(initial = 0, stiffness = 400, precision = 1e-6) {
  		this.current = initial;
  		this.target = initial;
  		this.#precision = precision;
  		this.configure(stiffness);
  	}
  
  	/**
  	 * Reconfigure spring stiffness. Call this when changing interaction modes.
  	 * Precomputes derived values for hot-path performance.
  	 *
  	 * @param {number} stiffness - New stiffness value
  	 */
  	configure(stiffness) {
  		this.#omega = Math.sqrt(stiffness);
  		this.#expDt = Math.exp(-this.#omega * DT);
  	}
  
  	/**
  	 * Advance physics by one fixed timestep.
  	 * Uses analytical solution for critically-damped spring.
  	 *
  	 * This is the "authoritative" state update - call at fixed rate (e.g., 360Hz).
  	 * Uses precomputed constants for maximum performance.
  	 */
  	step() {
  		const omega = this.#omega;
  		const expDt = this.#expDt;
  		const d = this.current - this.target;
  		const v = this.velocity;
  
  		// Analytical solution for critically-damped spring:
  		// x(t) = target + e^(-ωt) * (d + (v + ωd)t)
  		// v(t) = e^(-ωt) * (v + ωd) - ω * x(t)
  		const displacement = expDt * (d + (v + omega * d) * DT);
  		this.current = this.target + displacement;
  		this.velocity = displacement * -omega + expDt * (v + omega * d);
  	}
  
  	/**
  	 * Get interpolated render value without mutating state.
  	 * Call this once per frame with the remaining accumulator time.
  	 *
  	 * This provides smooth visuals even when physics rate differs from display rate.
  	 *
  	 * @param {number} dt - Time to extrapolate (in seconds), typically the accumulator remainder
  	 * @returns {number} Extrapolated position for rendering
  	 */
  	getRenderable(dt) {
  		const d = this.current - this.target;
  		const v = this.velocity;
  
  		// Early exit if at rest
  		if (d === 0 && v === 0) return this.current;
  
  		// Early exit if no extrapolation needed
  		if (dt === 0) return this.current;
  
  		// Same analytical solution, but for arbitrary dt (not precomputed)
  		const omega = this.#omega;
  		const exp = Math.exp(-omega * dt);
  		return this.target + exp * (d + (v + omega * d) * dt);
  	}
  
  	/**
  	 * "Grab" the spring - used when starting a drag interaction.
  	 * Snaps target to current position and zeros velocity.
  	 * This prevents the spring from fighting the user's input.
  	 */
  	grab() {
  		this.target = this.current;
  		this.velocity = 0;
  	}
  
  	/**
  	 * Set target value. The spring will animate toward this value.
  	 *
  	 * @param {number} value - New target value
  	 */
  	setTarget(value) {
  		this.target = value;
  	}
  
  	/**
  	 * Instantly set both current and target to value, with zero velocity.
  	 * Use for initialization or hard resets (no animation).
  	 *
  	 * @param {number} value - Value to set
  	 */
  	set(value) {
  		this.current = value;
  		this.target = value;
  		this.velocity = 0;
  	}
  
  	/**
  	 * Check if spring is at rest (close enough to target with negligible velocity).
  	 * Used to determine when to stop animating.
  	 *
  	 * @param {number} [precision] - Optional override for precision threshold
  	 * @returns {boolean} True if spring is at rest
  	 */
  	isAtRest(precision) {
  		const p = precision ?? this.#precision;
  		return Math.abs(this.current - this.target) < p && Math.abs(this.velocity) < p;
  	}
  
  	/**
  	 * Get current omega (natural frequency). Useful for debugging.
  	 * @returns {number}
  	 */
  	get omega() {
  		return this.#omega;
  	}
  }

  	}

  		if (doY && this.scaleTypes?.y === 'linear' && this.#yScaleInstance) {

  				this.layout.innerWidth,
  				false
  			);
  			this.#xScaleInstance.accumulateZoom(dataPoint, scaleFactor);
  			this.#xScaleInstance.applyZoomDomain(dataPoint, scaleFactor, 'x', this.layout.innerWidth);

  		if (doY && this.scaleTypes?.y === 'linear' && this.#yScaleInstance) {

  import { setContext, getContext, untrack } from 'svelte';
  import { LinearScaleState } from './LinearScaleState.svelte.js';
  import { BandScaleState } from './BandScaleState.svelte.js';
  import { scaleOrdinal } from 'd3-scale';
  
  // Helper to stringify sorted
  const stableStringify = (obj) => {
  	if (typeof obj !== 'object' || obj === null) return JSON.stringify(obj);
  	if (Array.isArray(obj)) return `[${obj.map(stableStringify).join(',')}]`;
  	// Sort keys to ensure consistent ID regardless of prop order
  	return `{${Object.keys(obj)
  		.sort()
  		.map((k) => `${JSON.stringify(k)}:${stableStringify(obj[k])}`)
  		.join(',')}}`;
  };
  
  class ScaleRegistry {
  	// 1. PUBLIC field (no #) to fix the "undefined" error
  	scaleUsage = $state(new Map());
  	#gpuCoordinator = null;
  
  	constructor() {
  		this.registryId = Math.random().toString(36).slice(2, 8);
  
  		// Initialize the storage buckets
  		this.scales = {
  			linear: new Map(),
  			band: new Map(),
  			ordinal: new Map()
  		};
  	}
  
  	setGPUCoordinator(coordinator) {
  		this.#gpuCoordinator = coordinator;
  	}
  
  	#notifyTopologyChange() {
  		if (this.#gpuCoordinator) {
  			this.#gpuCoordinator.scheduleTopologyUpdate();
  		}
  	}
  
  	getOrCreateScale(name, type = 'linear', config = {}, chartId = null) {
  		// const configKey = Object.keys(config).length > 0 ? `-${JSON.stringify(config)}` : '';
  		const configKey = Object.keys(config).length > 0 ? `-${stableStringify(config)}` : '';
  		const key = `${name}-${type}${configKey}`;
  
  		// 2. Dynamic Map creation (fixes the crash on 'log', 'point', etc.)
  		if (!this.scales[type]) {
  			this.scales[type] = new Map();
  		}
  
  		if (!this.scales[type].has(key)) {
  			// Factory Logic
  			if (['linear', 'log', 'time', 'pow', 'identity'].includes(type)) {
  				this.scales[type].set(
  					key,
  					new LinearScaleState(name, undefined, type, undefined, config, this.#gpuCoordinator)
  				);
  			} else if (['band', 'point'].includes(type)) {
  				this.scales[type].set(key, new BandScaleState(name, [], config));
  			} else if (['ordinal', 'categorical'].includes(type)) {
  				this.scales[type].set(
  					key,
  					scaleOrdinal()
  						.range(['#EFB605', '#FFF84A', '#FF0266'])
  						.domain(['Adelie', 'Gentoo', 'Chinstrap'])
  				);
  			} else {
  				// Fallback
  				this.scales[type].set(
  					key,
  					new LinearScaleState(name, undefined, 'linear', undefined, config, this.#gpuCoordinator)
  				);
  			}
  		}
  
  		const scale = this.scales[type].get(key);
  
  		if (chartId && scale) {
  			// 3. STOP THE LOOP: Untrack the bookkeeping!
  			untrack(() => {
  				if (!this.scaleUsage.has(scale)) {
  					this.scaleUsage.set(scale, new Set());
  				}
  
  				const chartIds = this.scaleUsage.get(scale);
  				// Only write if necessary (optimization)
  				if (!chartIds.has(chartId)) {
  					chartIds.add(chartId);
  					this.#notifyTopologyChange();
  				}
  			});
  		}
  
  		return scale;
  	}
  
  	deregisterChart(scale, chartId) {
  		if (!scale || !chartId) return;
  
  		// 4. STOP THE LOOP: Untrack cleanup too
  		untrack(() => {
  			if (this.scaleUsage.has(scale)) {
  				const usage = this.scaleUsage.get(scale);
  				if (usage.delete(chartId)) {
  					if (usage.size === 0) {
  						if (scale.destroy) scale.destroy();
  						this.scaleUsage.delete(scale);
  						this._removeScaleFromStorage(scale);
  					}
  					this.#notifyTopologyChange();
  				}
  			}
  		});
  	}
  
  	removeChartUsage(chartId) {
  		if (!chartId) return;
  
  		// 🛡️ UNTRACK is vital here to prevent the 'Read -> Track -> Write -> Trigger' cycle
  		untrack(() => {
  			let changed = false;
  
  			// Iterate (Read)
  			for (const [scale, usageSet] of this.scaleUsage) {
  				if (usageSet.has(chartId)) {
  					// Mutate (Write)
  					usageSet.delete(chartId);
  					changed = true;
  
  					if (usageSet.size === 0) {
  						if (scale.destroy) scale.destroy();
  						this.scaleUsage.delete(scale); // Mutate (Write)
  						this._removeScaleFromStorage(scale);
  					}
  				}
  			}
  
  			if (changed) {
  				this.#notifyTopologyChange();
  			}
  		});
  	}
  
  	// ... removeChartUsage, _removeScaleFromStorage (copy from previous if needed, or keeping it clean)
  	_removeScaleFromStorage(scaleInstance) {
  		for (const map of Object.values(this.scales)) {
  			for (const [key, scale] of map) {
  				if (scale === scaleInstance) {
  					map.delete(key);
  					return;
  				}
  			}
  		}
  	}
  }
  
  export function setScaleRegistry() {
  	const state = new ScaleRegistry();
  	return setContext('scale-registry', state);
  }
  
  export function getScaleState(name, type = 'linear', config = {}, chartId = null) {
  	const scaleRegistry = getContext('scale-registry');
  	if (!scaleRegistry) {
  		console.error('No Registry');
  		return null;
  	}
  	return scaleRegistry.getOrCreateScale(name, type, config, chartId);
  }

  	async function initWebGPU() {
  		try {
  			// Get shared GPU coordinator
  			gpuCoordinator = getGPUContext();
  
  			if (!gpuCoordinator) {
  				throw new Error(
  					'GPUCoordinator not found. Ensure setCoordinators() was called in a parent component.'
  				);
  			}
  
  			// Wait for coordinator initialization with timeout
  			let attempts = 0;
  			const maxAttempts = 100; // 5 seconds at 50ms intervals
  
  			while (!gpuCoordinator.isInitialized && attempts < maxAttempts) {
  				await new Promise((resolve) => setTimeout(resolve, 50));
  				attempts++;
  			}
  
  			if (!gpuCoordinator.isInitialized) {
  				throw new Error(`GPUCoordinator failed to initialize within ${maxAttempts * 50}ms timeout`);
  			}
  
  			// Configure context with proper error handling
  			await configureContext();
  
  			// Register with coordinator after context is ready
  			registerWithCoordinator();
  
  			console.log(`✅ WebGPUCanvas ${chartId} initialized successfully`);
  		} catch (error) {
  			console.error(`❌ WebGPUCanvas ${chartId} initialization failed:`, error);
  			initializationError = error.message;
  			throw error;
  		}
  	}

  	async function configureContext() {
  		if (!canvas) {
  			throw new Error('Canvas element not available for context configuration');
  		}
  
  		try {
  			// Get WebGPU context
  			context = canvas.getContext('webgpu');
  			if (!context) {
  				throw new Error('Failed to get WebGPU context from canvas. WebGPU may not be supported.');
  			}
  
  			const device = gpuCoordinator.device;
  			const presentationFormat = gpuCoordinator.presentationFormat;
  
  			if (!device) {
  				throw new Error('No WebGPU device available from coordinator');
  			}
  
  			// Configure context
  			context.configure({
  				device,
  				format: presentationFormat,
  				alphaMode: 'premultiplied',
  				desynchronized: true
  				// desynchronized: false
  			});
  
  			contextConfigured = true;
  
  			// Setup initial canvas size
  			setupCanvas();
  		} catch (error) {
  			console.error(`❌ Context configuration failed for chart ${chartId}:`, error);
  			contextConfigured = false;
  			throw error;
  		}
  	}

  		if (!contextConfigured) {
  			console.error(`⚠️ Attempting to register chart ${chartId} before context is configured`);

  		}
  
  		console.log(`📋 Registering chart ${chartId} with coordinator`);

  		if (!canvas || !contextConfigured) {

  			return;
  		}
  
  		try {
  			// Calculate new dimensions
  			const devicePixelRatio = window.devicePixelRatio || 1;
  
  			const canvasWidth = Math.ceil(width * devicePixelRatio);
  			const canvasHeight = Math.ceil(height * devicePixelRatio);
  
  			// Only update if dimensions actually changed to avoid unnecessary work
  			const dimensionsChanged = canvas.width !== canvasWidth || canvas.height !== canvasHeight;
  
  			const newTransform = `translate(${margin.left}px, ${margin.top}px)`;
  			const positionChanged = lastTransform !== newTransform;
  
  			// Only proceed if there's an actual change
  			if (!dimensionsChanged && !positionChanged) return;
  
  			if (dimensionsChanged) {
  				canvas.width = canvasWidth;
  				canvas.height = canvasHeight;
  				canvas.style.width = width + 'px';
  				canvas.style.height = height + 'px';
  			}

  				canvas.style.transform = newTransform;

  			}

  			}

  			console.error(`❌ Canvas setup failed for ${chartId}:`, error);

  // BaseGPURenderer.svelte.js - Registry Pattern Implementation
  // Scale management delegated to coordinator - renderer just identifies itself
  
  export class BaseGPURenderer {
  	// Core properties
  	#markType;
  	#webGPUContext;
  	#gpuCoordinator;
  	#chartContext;
  
  	// Identity & Async Control
  	#rendererId;
  	#initGeneration = 0;
  	#isDestroyed = false;
  
  	// Resource tracking (Exclusive resources only)
  	#ownedResources = {
  		buffers: [],
  		layerControl: null
  	};
  
  	// State management (Reactive - drives UI)
  	#resourcesCreated = $state(false);
  	#initializationError = $state(null);
  	#instanceCount = $state(0);
  
  	// GPU Objects
  	#device = null;
  	#pipeline = null;
  	#perDataBindGroup = $state(null);
  	#perDataLayout = null;
  
  	#dataBindingConfig = null;
  	#zIndex;
  	#xScaleType;
  	#yScaleType;
  
  	constructor(markType, webGPUContext, gpuCoordinator, chartContext, options = {}) {
  		this.#markType = markType;
  		this.#webGPUContext = webGPUContext;
  		this.#gpuCoordinator = gpuCoordinator;
  		this.#chartContext = chartContext;
  
  		// Unique ID for the Resource Registry
  		this.#rendererId = `${markType}-${crypto.randomUUID()}`;
  
  		this.#zIndex = options.zIndex || 0;
  		this.#xScaleType = options.xScaleType || 'linear';
  		this.#yScaleType = options.yScaleType || 'linear';
  	}
  
  	// ===== PUBLIC GETTERS =====
  
  	get resourcesCreated() {
  		return this.#resourcesCreated;
  	}
  	get initializationError() {
  		return this.#initializationError;
  	}
  	get instanceCount() {
  		return this.#instanceCount;
  	}
  	get pipeline() {
  		return this.#pipeline;
  	}
  	get markType() {
  		return this.#markType;
  	}
  	get perDataBindGroup() {
  		return this.#perDataBindGroup;
  	}
  
  	// ===== INITIALIZATION =====
  
  	async initialize(config) {
  		if (this.#isDestroyed) return;
  
  		// Async Race Protection: Increment generation
  		// Any previous pending initialization will see mismatching generation and abort
  		const currentGen = ++this.#initGeneration;
  
  		// If re-initializing, clean up previous shared resources via Registry
  		if (this.#resourcesCreated) {
  			this.#gpuCoordinator.releaseRenderer(this.#rendererId);
  			this.#resourcesCreated = false;
  		}
  
  		try {
  			const { dataBindingConfig, markShaderCode, vertexBufferLayout } = config;
  
  			if (!dataBindingConfig) {
  				throw new Error(`${this.#markType} renderer requires dataBindingConfig`);
  			}
  
  			this.#device = this.#webGPUContext.getDevice();
  			if (!this.#device) {
  				throw new Error('No WebGPU device available from context');
  			}
  
  			this.#dataBindingConfig = dataBindingConfig;
  
  			// ACQUIRE VIA REGISTRY
  			// Coordinator handles keying, caching, and ref-counting
  			// We just identify ourselves with rendererId
  			const result = await this.#gpuCoordinator.acquirePipeline(
  				this.#rendererId,
  				this.#markType,
  				{
  					x: this.#chartContext.scaleTypes?.x || this.#xScaleType,
  					y: this.#chartContext.scaleTypes?.y || this.#yScaleType
  				},
  				markShaderCode,
  				vertexBufferLayout,
  				dataBindingConfig
  			);
  
  			// POST-AWAIT SAFETY CHECK
  			// If destroyed or a newer init call happened while we were awaiting, abort and cleanup
  			if (this.#isDestroyed || this.#initGeneration !== currentGen) {
  				console.warn(`${this.#markType}: Stale initialization detected, discarding.`);
  				this.#gpuCoordinator.releaseRenderer(this.#rendererId);
  				return;
  			}
  
  			this.#pipeline = result.pipeline;
  			this.#perDataLayout = result.dataLayout;
  			this.#resourcesCreated = true;
  
  			console.log(`${this.#markType} renderer initialized (${this.#rendererId})`);
  		} catch (error) {
  			if (!this.#isDestroyed && this.#initGeneration === currentGen) {
  				console.error(`${this.#markType} initialization failed:`, error);
  				this.#initializationError = error.message;
  				throw error;
  			}
  		}
  	}
  
  	// ===== BIND GROUP CREATION =====
  
  	createDataBindGroup(dataBindings) {
  		if (!this.#device || !this.#perDataLayout) {
  			console.warn(
  				`Cannot create per-data bind group for ${this.#markType} - missing device or layout`
  			);
  			return;
  		}
  
  		if (!Array.isArray(dataBindings) || dataBindings.length === 0) {
  			console.warn(
  				`Cannot create per-data bind group for ${this.#markType} - invalid dataBindings`
  			);
  			return;
  		}
  
  		try {
  			// Validate that all required bindings are provided
  			const expectedBindings = this.#dataBindingConfig.map((config) => config.binding);
  			const providedBindings = dataBindings.map((binding) => binding.binding);
  
  			for (const expectedBinding of expectedBindings) {
  				if (!providedBindings.includes(expectedBinding)) {
  					throw new Error(`Missing data binding ${expectedBinding} for ${this.#markType}`);
  				}
  			}
  
  			this.#perDataBindGroup = this.#device.createBindGroup({
  				label: `${this.#markType}-per-data-bind-group`,
  				layout: this.#perDataLayout,
  				entries: dataBindings
  			});
  
  			console.debug(
  				`Per-data bind group (@group(1)) created for ${this.#markType} with ${dataBindings.length} bindings`
  			);
  		} catch (error) {
  			console.error(`Failed to create per-data bind group for ${this.#markType}:`, error);
  			this.#initializationError = error.message;
  			throw error;
  		}
  	}
  
  	// ===== BUFFER MANAGEMENT (Exclusive Resources) =====
  
  	createOrUpdateBuffer(
  		existingBuffer,
  		data,
  		usage = GPUBufferUsage.STORAGE | GPUBufferUsage.COPY_DST
  	) {
  		if (!this.#device) {
  			throw new Error('No WebGPU device available');
  		}
  
  		const requiredSize = data.byteLength;
  
  		// Smart buffer resizing to prevent thrashing
  		if (!existingBuffer || existingBuffer.size < requiredSize) {
  			if (existingBuffer) {
  				const index = this.#ownedResources.buffers.indexOf(existingBuffer);
  				if (index !== -1) {
  					this.#ownedResources.buffers.splice(index, 1);
  				}
  				existingBuffer.destroy();
  			}
  
  			// Smart sizing with headroom
  			const MEMORY_BLOCK_SIZE = 256;
  			const desiredSize = Math.ceil(requiredSize * 1.5);
  			const roundedUpSize = Math.ceil(desiredSize / MEMORY_BLOCK_SIZE) * MEMORY_BLOCK_SIZE;
  			let bufferSize = Math.max(requiredSize, roundedUpSize);
  			bufferSize = (bufferSize + 3) & ~3; // 4-byte alignment
  
  			const newBuffer = this.#device.createBuffer({
  				size: bufferSize,
  				usage,
  				label: `${this.#markType}-data-buffer-${this.#ownedResources.buffers.length}`
  			});
  
  			this.#ownedResources.buffers.push(newBuffer);
  			console.debug(
  				`Created ${this.#markType} buffer of size ${bufferSize} bytes (required: ${requiredSize})`
  			);
  			return newBuffer;
  		}
  
  		return existingBuffer;
  	}
  
  	updateInstanceCount(count) {
  		this.#instanceCount = count;
  		this.invalidateLayer();
  	}
  
  	registerLayer(drawFunction) {
  		try {
  			const chartId = this.#chartContext.chartId;
  
  			const layerConfig = {
  				draw: drawFunction,
  				markType: this.#markType,
  				renderer: this,
  				componentId: `${this.#markType}-${crypto.randomUUID().slice(0, 6)}`
  			};
  
  			this.#ownedResources.layerControl = this.#gpuCoordinator.registerLayer(chartId, layerConfig);
  
  			console.log(`${this.#markType} layer registered with coordinator for chart ${chartId}`);
  			return this.#ownedResources.layerControl;
  		} catch (error) {
  			console.error(`${this.#markType} layer registration failed:`, error);
  			this.#initializationError = error.message;
  			throw error;
  		}
  	}
  
  	invalidateLayer() {
  		if (this.#gpuCoordinator && this.#chartContext?.chartId) {
  			this.#gpuCoordinator.invalidateChart(this.#chartContext.chartId);
  		}
  	}
  
  	// ===== DATA PROCESSING HELPERS =====
  
  	processDataForAxis(data, axis) {
  		const scaleType = axis === 'x' ? this.#xScaleType : this.#yScaleType;
  		const scaleState =
  			axis === 'x' ? this.#chartContext.xScaleState : this.#chartContext.yScaleState;
  
  		switch (scaleType) {
  			case 'linear':
  			case 'identity':
  				return new Float32Array(data.map((d) => d ?? 0));
  
  			case 'band':
  			case 'point':
  				return scaleState.processDataForGPU(data, this.#markType);
  
  			case 'log':
  				return new Float32Array(data.map((d) => Math.log(Math.max(d || 1, 1))));
  
  			case 'time':
  				return new Float32Array(data.map((d) => (d instanceof Date ? d.getTime() : d || 0)));
  
  			default:
  				console.warn(`Unknown scale type: ${scaleType}, treating as linear`);
  				return new Float32Array(data.map((d) => d ?? 0));
  		}
  	}
  
  	// ===== UTILITY METHODS =====
  
  	parseColor(color) {
  		try {
  			if (typeof color === 'string' && color.startsWith('#')) {
  				const r = parseInt(color.slice(1, 3), 16) / 255;
  				const g = parseInt(color.slice(3, 5), 16) / 255;
  				const b = parseInt(color.slice(5, 7), 16) / 255;
  				return [r, g, b, 1.0];
  			}
  			console.warn(`Unrecognized color format: ${color}, using default blue`);
  			return [0.0, 0.0, 1.0, 1.0];
  		} catch (error) {
  			console.error(`Color parsing failed for ${this.#markType}:`, error);
  			return [0.0, 0.0, 1.0, 1.0];
  		}
  	}
  
  	// ===== CLEANUP =====
  
  	destroy() {
  		this.#isDestroyed = true;
  		// Bump generation so any pending init aborts
  		this.#initGeneration++;
  
  		console.log(`Cleaning up ${this.#markType} renderer (${this.#rendererId})`);
  
  		try {
  			// 1. Clean up EXCLUSIVE resources (buffers owned only by this renderer)
  			this.#ownedResources.buffers.forEach((buffer) => {
  				if (buffer) buffer.destroy();
  			});
  			this.#ownedResources.buffers = [];
  
  			// 2. Clean up layer registration
  			if (this.#ownedResources.layerControl) {
  				this.#ownedResources.layerControl.remove();
  				this.#ownedResources.layerControl = null;
  			}
  
  			// 3. Clean up SHARED resources via Registry
  			// Coordinator tracks what we acquired and decrements ref-counts
  			if (this.#gpuCoordinator) {
  				this.#gpuCoordinator.releaseRenderer(this.#rendererId);
  			}
  
  			// Clear references
  			this.#pipeline = null;
  			this.#perDataBindGroup = null;
  			this.#perDataLayout = null;
  			this.#device = null;
  
  			console.log(`${this.#markType} renderer cleanup complete`);
  		} catch (error) {
  			console.error(`Error during ${this.#markType} renderer cleanup:`, error);
  		}
  	}
  }

   * A generic controller that handles the common lifecycle, state management,
   * and data fetching orchestration for a mark component using reactive patterns.
   * This class acts as the "ViewModel" or "Controller" for a mark.
   */
  export class BaseMarkSetup {
    // --- Reactive State (for the View to consume) ---
    client = $state(null);
    categoricalDataFetched = $state(false);
  
    // --- Reactive Setup Progress State ---
    #scaleStatesReady = $state(false);
    #scaleMappingsReady = $state(false);
    #clientCreated = $state(false);
    #clientDataReady = $state(false);
    #continuousDomainsSet = $state(false);

    // --- Private Properties ---
    #config;
    #coordinator;
    #chartContext;
    #markId;
    #scaleMap = {};
  
    /**
     * @param {object} config The mark's configuration object, derived from Svelte props.
     * @param {import('@uwdata/mosaic-core').Coordinator} coordinator The Mosaic coordinator instance.
     * @param {object} chartContext The chart's context object.
     */
    constructor(config, coordinator, chartContext, stableId = null) {
      this.#config = config;
      this.#coordinator = coordinator;
      this.#chartContext = chartContext;
      this.#markId =  stableId || `${config.markType}-${crypto.randomUUID().slice(0, 8)}`;
      this.#registerScaleRequirements();
      this.#setupReactiveEffects();
    }
  
    #setupReactiveEffects() {
  
      // Effect 1: Wait for scale objects to be available in context
      $effect(() => {
        const { scaleTypes, xScaleState, yScaleState } = this.#chartContext;
        if (!!(scaleTypes && xScaleState && yScaleState) && !this.#scaleStatesReady) {
          this.#scaleStatesReady = true;
        }
      });
  
      // Effect 2: Setup scale mappings when scale states are ready
      $effect(() => {
        if (this.#scaleStatesReady && !this.#scaleMappingsReady) {
          this.#setupScaleMappings();
          this.#scaleMappingsReady = true;
        }
      });
  
      // Effect 3: Create client when scale mappings are ready
      $effect(() => {
        if (this.#scaleMappingsReady && !this.#clientCreated) {
          const fieldRequests = this.#buildFieldInfoRequests();
          this.#createClient(fieldRequests);
          this.#clientCreated = true;
        }
      });
  
      // Effect 4: Monitor client data readiness
      $effect(() => {
        if (this.#clientCreated && this.client?.fieldInfo && !this.#clientDataReady) {
          this.#clientDataReady = true;
        }
      });
  
      // Effect 5: Set initial continuous domains when client data is ready
      $effect(() => {
        if (this.#clientDataReady && !this.#continuousDomainsSet) {
          this.#setInitialContinuousDomains(this.client.fieldInfo);
          this.#continuousDomainsSet = true;
        }
      });
  
      // Effect 6: Start categorical domain fetch when continuous domains are set
      $effect(() => {
        if (this.#continuousDomainsSet && !this.#categoricalFetchStarted) {
          this.#categoricalFetchStarted = true;
          
          // Trigger async operation without awaiting in effect
          this.#fetchCategoricalDomains().then(() => {
            this.categoricalDataFetched = true;

     * @virtual
     * Builds the SQL query for the DataClient.

    buildQuery(sourceTable, config, filter) {
      const { select, sort, limit, _filterNulls } = config;
  
      const q = Query.from(sourceTable);
      const selectDef = {};
      const nonAggregateGroupBy = [];
      let hasAggregates = false;
  
      for (const alias in select) {
        const expr = select[alias];
        if (expr == null) continue;
  
        const exprNode = isNode(expr) ? expr : sqlColumn(String(expr));
        selectDef[alias] = exprNode;
  
        if (isAggregateExpression(exprNode)) {
          hasAggregates = true;
        } else {
          nonAggregateGroupBy.push(exprNode);
        }
      }
      q.select(selectDef);
  
      if (hasAggregates && nonAggregateGroupBy.length > 0) {
        q.groupby(nonAggregateGroupBy);
      }
  
      const allFilters = [...filter];
      if (_filterNulls && Array.isArray(_filterNulls)) {
        allFilters.push(..._filterNulls.map(col => isNotNull(col)));
      }
      if (allFilters.length > 0) {
        q.where(and(allFilters));
      }
      
      if (sort) q.orderby({ [sort.by]: sort.direction === 'desc' ? 'DESC' : 'ASC' });
      if (limit) q.limit(limit);
  
      return q;
    }

      this.client = new DataClient(
        this.#config.table,
        this.#config.selection,
        {
          select: this.getSelectConfig(),
          queryBuilder: (table, config, filter) => this.buildQuery(table, config, filter),
          dataFormat: 'arrow',
          name: this.#markId,
          ...this.#config.clientOptions
        },
        fieldRequests
      );

        y: this.#chartContext.yScaleState
      };
      
      console.log(`📊 ${this.#markId} Scale mappings setup:`, {
        x: this.actualXScaleType,
        y: this.actualYScaleType
      });
    }
  
    #buildFieldInfoRequests() {
      const { select, table } = this.#config;
      const requests = [];
      const processedFields = new Set();
  
      for (const alias in select) {
        const fieldName = select[alias];
        if (fieldName == null || processedFields.has(fieldName.toString())) continue;
  
        const exprNode = isNode(fieldName) ? fieldName : sqlColumn(String(fieldName));
        if (isAggregateExpression(exprNode)) continue;
  
        const scaleType = alias === 'x' ? this.actualXScaleType : alias === 'y' ? this.actualYScaleType : null;
  
        if (scaleType === 'linear' || scaleType === 'time' || scaleType === 'log') {
          requests.push({ table, column: exprNode, stats: ['min', 'max'] });
          processedFields.add(fieldName.toString());
        }
      }
      return requests;
    }
    
    #setInitialContinuousDomains(fieldInfo) {
      if (!fieldInfo) return;
  
      for (const alias in this.#scaleMap) {
        const scaleState = this.#scaleMap[alias];
        if (!scaleState) continue;
  
        const scaleType = alias === 'x' ? this.actualXScaleType : this.actualYScaleType;
        
        if (scaleType === 'linear' || scaleType === 'time' || scaleType === 'log') {
          const fieldName = this.#config.select[alias];
          if (fieldName == null) continue;
  
          const infoKey = String(fieldName).replace(/^["']|["']$/g, '');
          const info = fieldInfo[infoKey];
  
          if (info && info.min != null && info.max != null) {
            scaleState.addMark([info.min, info.max], this.#markId);
          }
        }
      }
    }
  
    async #fetchCategoricalDomains() {
      const { table, selection, sort } = this.#config;
      if (!this.#coordinator || !table) return;
  
      const fetches = [];
  
      const setupFetch = (axis, field) => {
        if (!field) return;
        const scaleState = this.#chartContext[`${axis}ScaleState`];
        let q = Query.from(table).select({ [axis]: sqlColumn(String(field)) });
  
        // Handle sorting for categorical domains
        const sortConfig = sort;
        if (sortConfig && sortConfig.by) {
          const isSortingByOtherAxis = sortConfig.by === (axis === 'x' ? 'y' : 'x');
          const otherAxisField = this.#config.select[sortConfig.by];
          if (isSortingByOtherAxis && isAggregateExpression(otherAxisField)) {
            q = q.groupby(field).select({ [sortConfig.by]: otherAxisField });
          }
          q.orderby({ [sortConfig.by]: sortConfig.direction === 'desc' ? 'DESC' : 'ASC' });
        } else {
          q.distinct();
        }
  
        fetches.push(
          this.#coordinator.query(q).then(result => {
            const categories = Array.from(result).map(row => row[axis]);
            if (categories.length > 0) {
              scaleState.addMark(categories, this.#markId);
            }
          })
        );
      };
  
      if (this.actualXScaleType === 'band' || this.actualXScaleType === 'point') {
        setupFetch('x', this.#config.select.x);
      }
      if (this.actualYScaleType === 'band' || this.actualYScaleType === 'point') {
        setupFetch('y', this.#config.select.y);
      }
  
      if (fetches.length > 0) await Promise.all(fetches);
    }
  
    #setScalesReady() {
      Object.values(this.#scaleMap).forEach(scale => scale?.setReady?.());
    }
  }
